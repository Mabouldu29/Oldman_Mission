// When player has equipment, tracker will be beeping
BIS_ctrgCaches = [];
BIS_hasDetector = true;
BIS_OM_trackerEnabled = TRUE;

#define BEEP_RANGE 300

BIS_fnc_om_trackerAgent = {
	scriptName "BIS_fnc_om_trackerAgent";
	while {true} do {
		sleep 2;
		PROFILE_INFO("BIS_fnc_om_trackerAgent","BIS_fnc_om_trackerAgent",1);
		private _nearbyCaches = BIS_ctrgCaches select {call (_x getVariable ["CustomCondition", {TRUE}]) && !isNull _x && {alive _x} && {!GVAR(_x,"discovered")} && {_x distance player < (_x getVariable "range") + 20}};
		private _detectorArr = (missionNamespace getVariable ["BIS_OM_PhoneMenuBox", []]) findIf {(_x # 0) == "Detector"};
		private _detectorAdded = FALSE;
		if (_detectorArr != -1) then {
			_detectorAdded = (BIS_OM_PhoneMenuBox # _detectorArr) # 6;
		};
		if (BIS_hasDetector && count _nearbyCaches > 0 && _detectorAdded) then {
			
			if (BIS_OM_trackerEnabled) then {
				(findDisplay 46 displayCtrl 9999015) ctrlSetText "\a3\Missions_F_Oldman\Data\img\PhoneCore\mapIcon.pac";
			} else {
				(findDisplay 46 displayCtrl 9999015) ctrlSetText "\a3\Missions_F_Oldman\Data\img\PhoneCore\newmap.pac";
			};
			(findDisplay 46 displayCtrl 9999015) ctrlShow TRUE;
			private _handle = [_nearbyCaches] spawn BIS_fnc_om_beepInArea;
			waitUntil {sleep 0.5; scriptDone _handle};
		} else {
			(findDisplay 46 displayCtrl 9999015) ctrlShow FALSE;
		};
	};
};

// Basic beeping device
BIS_fnc_om_beepInArea = {
	scriptName "BIS_fnc_om_beepInArea";
	params ["_caches"];
	// Execute code when near cache
	{
		if (GVARS(_x,"InRangeCode","") != "") then {
			[] call compile GVARS(_x,"InRangeCode","");
		};
	} forEach _caches;

	while {_caches findIf {alive _x && {player distance _x < (_x getVariable "range")}} > -1 && BIS_OM_trackerEnabled} do {
		private _minDir = 180;
		private _object = objNull;
		{
			PROFILE_INFO("BIS_fnc_om_beepInAreaWhile","BIS_fnc_om_beepInAreaWhile",1);
			private _relDir = player getRelDir _x;
			if (_relDir > 180) then {
				_relDir = abs (360 - _relDir);
			};
			if (_relDir < _minDir) then {
				_minDir = _relDir;
				_object = _x;
			};

			// Cache was found, remove beeping from it
			if (_x distance player <= GVAR(_x,"discoveryRange")) then {
				SVAR(_x,"discovered",true);
				_caches = _caches - [_x];
				// Add marker to position

				sleep 1;
				playSound "TacticalPing2";

				/*private _mrk = createMarker [str _x, _x];
				_mrk setMarkerShape "ICON";
				_mrk setMarkerType "mil_end";
				_mrk setMarkerText "Supply Cache";
				_mrk setMarkerColor "ColorPink";*/

				if (_x getVariable ["ShowMarker", TRUE]) then {
					[0,["SmartMarker_I_CacheFound_F", format["BIS_cache%1", time]]] call BIS_fnc_OM_moduleSmartMarkers;
				};

				if (GVARS(_x,"FindCode","") != "") then {
					[] call compile GVARS(_x,"FindCode","");
				};
			};
		} forEach _caches;

		// Calculate range coeficient
		private _rangeCoef = 2 / GVAR(_object,"range") * (player distance _object) + 0.5;
		private _timeout = ((_minDir + 30) / 250) * _rangeCoef + 0.25;
		if (BIS_OM_trackerEnabled) then {
			playSound "FD_Timer_F";
		};
		sleep _timeout;
	};
};

BIS_fnc_om_initTrackers = {
	{
		private _range = GVAR(_x,"range");
		private _discoveryRange = GVAR(_x,"FindRange");
		private _findCode = GVARS(_x,"FindCode","");
		private _inRangeCode = GVARS(_x,"InRangeCode","");
		private _customCondition = compile GVARS(_x,"CustomCondition","TRUE");
		private _showMarker = GVARS(_x,"ShowMarker",TRUE);
		{
			SVAR(_x,"range",_range);
			SVAR(_x,"discoveryRange",_discoveryRange);
			SVAR(_x,"discovered",false);
			SVAR(_x,"InRangeCode",_inRangeCode);
			SVAR(_x,"FindCode",_findCode);
			SVAR(_x,"CustomCondition",_customCondition);
			SVAR(_x,"ShowMarker",_showMarker);
			BIS_ctrgCaches pushBackUnique _x;
		} forEach synchronizedObjects _x;
		deleteVehicle _x;
	} forEach (entities "ModuleOmTracked_F");
};
