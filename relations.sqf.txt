BIS_fnc_om_threatCalculation = {
	scriptName "BIS_fnc_om_threatCalculation";
	/* PROFILE_START("BIS_fnc_om_threatCalculation"); */
	params["_module", "_side"];
	PROFILE_START_INFO("BIS_fnc_om_threatCalculation","BIS_fnc_om_threatCalculation"+STR _side,1);
	private _vehicle = objectParent player;
	private _weapon = currentWeapon player;
	private	_finalRange = _module getVariable ["finalRange", RELATION_DEFAULT_RANGE_FOR_UNARMED_PLAYER];	// Default value for unarmed player
	private _finalRate = _module getVariable ["finalRate", RELATION_DEFAULT_RATE_FOR_UNARMED_PLAYER]; 		// Default value for unarmed player

	_processPlayerOnFoot = {
		private _typesAndCosts = [
			["PistolCore",50,4],
			["RifleCore",90,10],
			["MGunCore",110,15],
			["LauncherCore",120,20],
			["CannonCore",120,20]
		];
		{
			_x params ["_type","_range","_rate"];
			if (!(_weapon isEqualTo "") && {_weapon isKindOf [_type, configFile >> "CfgWeapons"]}) exitWith {
				_finalRange = _range;
				_finalRate = _rate;
			};
		} forEach _typesAndCosts;
	};

	if (_side == EAST && isNull _vehicle) then
	{
		call _processPlayerOnFoot;
		if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform) then
		{
			if (BIS_iswearingUndercoverFacewear) then {_finalRange = 6} else {_finalRange = 18};
			_finalRate = (_finalRate * 2) max 4;
		}
		else
		{
			if (BIS_isWearingCSATUniform) then
			{
				_finalRange = 30 max _finalRange;
				_finalRate = (_finalRate * 2) max 2.5;
			}
			else
			{
				if (!BIS_isWearingNeutralUniform) then
				{
					_finalRange = 150 max _finalRange;
					_finalRate = (_finalRate * 3) max 4;
				}
				else
				{
					if (BIS_isWearingSuspiciousVest) then
					{
						_finalRange = 40 max _finalRange;
						_finalRate = (_finalRate * 1.5) max 2;
					}
					else
					{
						if (BIS_isWearingSuspiciousHeadgear) then
						{
							_finalRange = 40 max _finalRange;
							_finalRate = (_finalRate * 2) max 2;
						}
						else
						{
							if (BIS_isWearingWorkerUniform || BIS_isWearingIDAPUniform) then
							{
								if (BIS_iswearingUndercoverFacewear) then {_finalRange = 6 max _finalrange/1.6} else {_finalRange};
								_finalRate = (_finalRate * 2) max 2;
							};
						};
					};
				};
			};
		};
	};
	if (isNull _vehicle) then {
		if (_side != EAST) then {
			call _processPlayerOnFoot;

		};
	} else {
		private _vehicleSide = getNumber (configfile >> "CfgVehicles" >> typeOf _vehicle >> "side");
		if (_vehicleSide == 0 || _vehicleSide == 1 || _vehicleSide == 2) then {
			private _typesAndCosts = [
				["Car_F",50,5],
				["Tank",100,10],
				["Helicopter",200,20],
				["Plane",250,20]
			];
			{
			_x params ["_type","_range","_rate"];
				if (_vehicle isKindOf _type) exitWith {
					// Car driving outside road will generate +50% higher threat
					if (_forEachIndex <= 1 && speed _vehicle > 5) then {
						_range = _range * 1.5;
						_rate = _rate * 1.5;
					};
					_finalRange = _range;
					_finalRate = _rate;
				};
			} forEach _typesAndCosts;
		};
	};
	//debuglog str [_finalRange,_finalRate];
	PROFILE_STOP("BIS_fnc_om_threatCalculation");
	[_finalRange,_finalRate]
};

BIS_fnc_om_relationsAgent = {
	scriptName "BIS_fnc_om_relationsAgent";
	params ["_module"];

	private _side = GVAR(_module,"side");

	SVAR(_module,"threat",GVAR(_module,"DefaultRating"));
	SVAR(_module,"aggroMode",M_CALM);
	Sleep ( random [0.3, 0.5, 0.8] );	//run multiple instances in another periods (preven ticking all instance at the same time)
	while {true} do {
		PROFILE_INFO("BIS_fnc_om_relationsAgent","BIS_fnc_om_relationsAgent_"+str _side,1);
		sleep TICK_TIME;
		// Calculate default threat, which is modified by other events
		([_module, _side] call BIS_fnc_om_threatCalculation) params ["_threatRange","_threatRate"];
		private _threatLevel = GVAR(_module,"threat");
		private _aggroMode = GVAR(_module,"aggroMode");

		// If player is aiming at anyone, he's much more threatening

		if (side group cursorTarget == _side && !visiblemap) then {
			_threatRange = _threatRange * 3;
			_threatRate = _threatRate * 3;
		};


		// Find out if player is in restricted area
		private _areas = [player,_side] call BIS_fnc_om_inRestrictedAreas;
		if (count _areas > 0) then {
			private _addedRanges = (_areas apply {_x getVariable "ThreatRange"});
			private _addedThreats = (_areas apply {_x getVariable "ThreatRate"});
			_addedRanges sort false;
			_addedThreats sort false;
			if (_side == EAST) then
			{
				_threatRange = _threatRange + ((_addedRanges#0) * (if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform ) then {0.0} else {1}));
				_threatRate = _threatRate + ((_addedThreats#0) * (if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform ) then {0.0} else {1}));
			} else {
				_threatRange = _threatRange + ((_addedRanges#0) * (if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform) then {1} else {1}));
				_threatRate = _threatRate + ((_addedThreats#0) * (if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform) then {3} else {1}));
			};

		};

		// Test if enemy is in Aggro range
		private _inRestrictedZone = count _areas > 0 && !GVAR(_module,"IgnoreRestrictedZones");
		if ( ( GVAR(_module,"AggroMode") != M_ALLIED || _inRestrictedZone ) && { [_side,_threatRange] call BIS_fnc_om_isPlayerDetected} ) then {
			_threatLevel = _threatLevel + _threatRate;
		} else {
			// Stabilization of the score
			private _defaultThreat = GVAR(_module,"DefaultRating");
			if (_threatLevel > _defaultThreat) then {
				_threatLevel = _threatLevel - GVAR(_module,"Stabilization");
				_threatLevel = _threatLevel max _defaultThreat;
			};
			if (_threatLevel < _defaultThreat) then {
				_threatLevel = _threatLevel + GVAR(_module,"Stabilization");
				_threatLevel = _threatLevel min _defaultThreat;
			};
		};

		CLAMP(_threatLevel,GVAR(_module,"MinRating"),GVAR(_module,"MaxRating"));
		SVAR(_module,"threat",_threatLevel);
		SVAR(_module,"threatRange",_threatRange);
		SVAR(_module,"threatRate",_threatRate);

		// Switch aggro mode
		private _ally = GVAR(_module,"allyTreshold");
		private _threatened = GVAR(_module,"threatTreshold");
		private _agressive = GVAR(_module,"enemyTreshold");
		switch (true) do {
			case (_threatLevel >= _agressive && _aggroMode != M_AGRESSIVE): {[_module] call BIS_fnc_om_attackPlayer};
			case (_threatLevel >= _threatened && _threatLevel < _agressive && _aggroMode != M_PREPARE): {[_module] call BIS_fnc_om_prepareToAttack};
			case (_threatLevel > _ally && _threatLevel < _threatened && _aggroMode != M_CALM): {[_module] call BIS_fnc_om_goCalm};
			case (_threatLevel <= _ally && _aggroMode != M_ALLIED): {[_module] call BIS_fnc_om_goAllied};
		};
	};
};

#define MAX_MESSAGE_RANGE 50

BIS_fnc_om_attackPlayer = {
	scriptName "BIS_fnc_om_attackPlayer";
	params ["_module"];
	private _side = GVAR(_module,"side");
	PROFILE_INFO("BIS_fnc_om_attackPlayer", "BIS_fnc_om_attackPlayer"+STR _side, 1);
	if (DEBUG_MODE) then {systemChat format ["REL: %1 is hostile",_side]};
	/* [_module] call BIS_fnc_om_showAwarenessUI; */
	SVAR(_module,"aggroMode",M_AGRESSIVE);

	BIS_playerSide setFriend [_side,0];
	_side setFriend [BIS_playerSide,0];
	//attack message
	////////////////////////////////////
	private _inRestrictedArea = count ([player] call BIS_fnc_om_inRestrictedAreas) > 0;
	private _enemies = [_side,200] call BIS_fnc_om_awareEnemies;
	private _nearestEnemy = objNull;
	private _sortedEnemies = [_enemies,[],{player distance _x},"ASCEND"] call BIS_fnc_sortBy;

	//hint format ["%1\n%2\n%3",_side,_protocol,_speaker];
	if (count _sortedEnemies > 0) then
	{
		_nearestEnemy = _sortedEnemies#0;
		[_nearestEnemy,west] spawn BIS_fnc_OM_Alarm
	} else {
		[player,west] spawn BIS_fnc_OM_Alarm
	};

	private _protocol = 0;
	private _speaker = "";
	switch _side do
	{
		case east:
		{
			//TODO: CHANGE CONDITION
			if (_nearestenemy isKindOf "B_GEN_Commander_F" || _nearestenemy isKindOf "B_GEN_Soldier_F") exitwith {_protocol = 1;_speaker = "Gendarmerie"};
			//if (uniform _nearestEnemy in ["U_O_GEN_Soldier_F","U_O_GEN_Commander_F"]) exitwith {_protocol = 1;_speaker = "Gendarmerie"};
			_protocol = 0;
			_speaker = "Soldier";
		};

		case independent:
		{
			_protocol = 2;
			_speaker = "Syndikat";
		};
	};

	if (_nearestEnemy distance player > MAX_MESSAGE_RANGE) exitWith {debugLog "BIS_fnc_om_intruderMessage: Nearest enemy too far from player"};

	private _selectedMessage = [];
	// Player was revealed in Restricted zone
	if (_inRestrictedArea) then
	{
		// player in vehicle
		if (vehicle player != player) exitwith {_selectedMessage = selectRandom (BIS_om_VehicleAttackMessages select _protocol)};
		if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {_selectedMessage = selectRandom (BIS_om_AttackMessagesCSATuniform select _protocol)};
		_selectedMessage = selectRandom (BIS_om_RestrictedZoneAttack select _protocol);
	}
	else
	{
		if (vehicle player != player) then
		// player in vehicle
		{_selectedMessage = selectRandom (BIS_om_VehicleAttackMessages select _protocol)} else
		// player is infantry
		{
			if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {_selectedMessage = selectRandom (BIS_om_AttackMessagesCSATuniform select _protocol)};
			_selectedMessage = selectRandom (BIS_om_AttackMessages select _protocol)
		};
	};

	// Select message to be played
	_selectedMessage params ["_sound","_message"];
	//debugLog str [_nearestEnemy, player distance _nearestEnemy, _sound, _message];

	[_nearestEnemy, player] say3D [_sound, 300, 1,true];
	[_speaker,_message] call BIS_fnc_showSubtitle;
	///////////////////////////////////////
	private _enemyGroups = allGroups select {leader _x distance2D player < SPAWN_RANGE};
	{
		_x setCombatMode "RED";
	} forEach _enemyGroups;
};

BIS_fnc_om_intruderMessage = {
	params ["_enemies","_restrictedArea"];
	PROFILE_INFO("BIS_fnc_om_intruderMessage","BIS_fnc_om_intruderMessage",1);
	// Select nearest enemy
	private _nearestEnemy = objNull;
	private _sortedEnemies = [_enemies,[],{player distance _x},"ASCEND"] call BIS_fnc_sortBy;

	if (count _sortedEnemies > 0) then {
		_nearestEnemy = _sortedEnemies#0;
	};

	private _protocol = 0;
	private _speaker = "";

	switch (side group _nearestEnemy) do
	{
		case east:
		{
			//TODO: CHANGE CONDITION
			if (_nearestenemy isKindOf "B_GEN_Commander_F" || _nearestenemy isKindOf "B_GEN_Soldier_F") exitwith {_protocol = 1;_speaker = localize "str_a3_cfgfactionclasses_blu_gen_f0"}; // TODO: Localize
			//if (uniform _nearestEnemy in ["U_O_GEN_Soldier_F","U_O_GEN_Commander_F"]) exitwith {_protocol = 1;_speaker = "Gendarmerie"};
			_protocol = 0;
			_speaker = localize "str_a3_character_soldier";	//TODO: Localize
		};

		case independent:
		{
			_protocol = 2;
			_speaker = localize "str_a3_character_insurgent"; 	//TODO: Localize
		};
	};

	if (_nearestEnemy distance player > MAX_MESSAGE_RANGE) exitWith {debugLog "BIS_fnc_om_intruderMessage: Nearest enemy too far from player"};

	private _selectedMessage = [];
	if (_restrictedArea) then {
		//Night
		if (BIS_night) exitwith
		{
			if (vehicle player != player) then
			{
				// vehicle
				_selectedMessage = selectRandom (BIS_om_RestrictedNightVehicleMessage select _protocol);
			} else {
				// infantry
				if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {_selectedMessage = selectRandom (BIS_om_AttackMessagesCSATuniform select _protocol)};
				_selectedMessage = selectRandom (BIS_om_RestrictedNightMessage select _protocol);
			};
		};
		//vehicle
		if (vehicle player != player) exitwith
		{
			//if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {};
			_selectedMessage = selectRandom (BIS_om_RestrictedZoneWarning select _protocol);
		};
		//infantry
		if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {_selectedMessage = selectRandom (BIS_om_AttackMessagesCSATuniform select _protocol)};
		_selectedMessage = selectRandom (BIS_om_RestrictedZoneWarning select _protocol);
	} else {
		//vehicle
		if (vehicle player != player) exitwith {_selectedMessage = selectRandom (BIS_om_VehicleWarningMessages select _protocol);};
		// infantry unarmed
		if (currentweapon player != "") exitwith
		{
			if (BIS_isWearingCSATHelmet && BIS_isWearingCSATUniform)  exitwith {_selectedMessage = selectRandom (BIS_om_defaultWarningMessagesCSATuniform select _protocol)};
			_selectedMessage = selectRandom (BIS_om_WeaponRevealedMessages select _protocol);
		};
		_selectedMessage = selectRandom (BIS_om_defaultWarningMessages select _protocol);//BIS_om_WeaponRevealedMessages
	};

	// Select message to be played
	_selectedMessage params ["_sound","_message"];
	debugLog str [_nearestEnemy, player distance _nearestEnemy, _sound, _message];

	[_nearestEnemy, player] say3D [_sound, 300, 1,true];
	[_speaker,_message] call BIS_fnc_showSubtitle;
};
