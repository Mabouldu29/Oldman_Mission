BIS_fnc_OM_ModuleSyndikatAgent_returnGroupsToPool =
{
	scriptName "BIS_fnc_OM_ModuleSyndikatAgent_returnGroupsToPool";
  params [ "_data", ["_target", [0,0,0], [[]] ] ];
  /* ["Data: %1", _data] call BIS_fnc_LogFormat; */

  if(count _data > 0) then
  {
    private _arrOfUnits = [];
    { _arrOfUnits append (units _x) } foreach (_data#1);
    (str _arrOfUnits) call BIS_fnc_log;
    ["SYNDIKATAGENT: Returning %1 Syndikat units to the pool after the succesfully finished assault", count _arrOfUnits] call BIS_fnc_logFormat;
    {
      _x doMove _target;
      [_x] spawn
      {
				scriptName "BIS_fnc_OM_ModuleSyndikatAgent_returnGroupsToPool/Spawn1";
        private _unit = _this select 0;
        private _logic = MGVAR("BIS_OM_SyndikatAgent_logic");
        if( alive _unit ) then
        {
          WaitUntil{sleep 1; (player distance _unit) > DISTANCE_TO_DISAPPEAR};
          private _groupToAdd = ([_unit] call BIS_fnc_om_scanAndDelete)#1#0;
          (str _groupToAdd) call BIS_fnc_Log;
          private _basicGroup = GVAR(_logic, 'basicGroup');
          private _actualNumberOfUnits = [_basicGroup] call BIS_fnc_OM_ModuleSyndikatAgent_getNumberFromData;
          if( _actualNumberOfUnits < GVAR( _logic, 'poolMax') ) then
          {
            (_basicGroup#1) pushBack _groupToAdd;
            SVAR( _logic, "PoolInitNumber", _actualNumberOfUnits );
            SVAR( _logic, "PoolPower", [_basicGroup] call BIS_fnc_OM_ModuleSyndikatAgent_GetTeamPower );
          };
        };
      };
    } forEach _arrOfUnits;
  }
  else
  {
    //we are trying to get non existing groups (most likely the support was canceled prematurely), lets get back groups from the backup
    private _logic = MGVAR("BIS_OM_SyndikatAgent_logic");
    {
        //TODO: do it properly (we are counting groups, but in group could be more than one soldier)
        if( ([GVAR(_logic,"basicGroup")] call BIS_fnc_OM_ModuleSyndikatAgent_getNumberFromData) < GVAR( _logic, "poolMax" ) ) then
        {
          SYNDIKATPOOL_GROUPS pushBack _x;
        };
    } forEach (GVAR(_logic, "copyOfUnits"))#1;    //copy units back to Pool, but add them in case another units appeared in the pool meanwhile
    ["SYNDIKATAGENT: Returning %1 Syndikat units to the pool after the assault was canceled before it started", count ((GVAR(_logic, "copyOfUnits")) select 1)] call BIS_fnc_logFormat;
    SVAR( _logic, "copyOfUnits", nil);
    SVAR( _logic, "PoolInitNumber",  [GVAR(_logic,"basicGroup")] call BIS_fnc_OM_ModuleSyndikatAgent_getNumberFromData );
    SVAR( _logic, "PoolPower", [GVAR(_logic,"basicGroup")] call BIS_fnc_OM_ModuleSyndikatAgent_GetTeamPower );
  };
};

private _Init =
{
	scriptName "_Init";
  params [ ["_logic", objNull, [objNull] ]];

  #ifdef SYNDIKATPOOL_DEBUG
    if(isNil "BIS_SyndikatPool_Debug") then
    { BIS_SyndikatPool_Debug = SYNDIKATPOOL_DEBUG };
  #else
    if(isNil "BIS_SyndikatPool_Debug") then
    { BIS_SyndikatPool_Debug = false }
  #endif

	SMGVAR("BIS_OM_SyndikatAgent_logic", _logic);
  SVAR( _logic, "State", SYNDIKATAGENT_STATE_RUNNING);
  private _lastEventDate = [date, -1*MIN_TIME_AFTER_EVENT_FINISHED] call BIS_fnc_CalculateDateTime;
  SVAR( _logic, "LastEventDate", _lastEventDate );
	private _newDate = [date, (GVAR(_logic,"EventSelectionPeriod"))] call BIS_fnc_CalculateDateTime;
	SVAR( _logic, "NextEventDate", _newDate);
  SVAR( _logic, "poolIncomeTime", (GVAR( _logic, "poolIncomeTime"))*60 );
  SVAR( _logic, "questIDs", -1);

	//Initialize the Syndikat Pool
	private _basicGroup = synchronizedObjects _logic;
	if( count _basicGroup > 0 ) then
	{
		_basicGroup = [_basicGroup] call BIS_fnc_om_scanAndDelete;
	}
	else
	{
		//use the very basic group if any unit is synchronized to the module
		_basicGroup = SYNDIKATPOOL_BASIC_GROUP;
	};

  SVAR( _logic, "basicGroup", _basicGroup );
  SVAR( _logic, "PoolInitNumber", [_basicGroup] call BIS_fnc_OM_ModuleSyndikatAgent_getNumberFromData );
  SVAR( _logic, "PoolPower", [_basicGroup] call BIS_fnc_OM_ModuleSyndikatAgent_GetTeamPower );
	SVAR( _logic, "sectors", [] call BIS_fnc_OM_ModuleSyndikatAgent_BuildSectorsData );
  [] spawn _DumpSectorsInfo;
  _logic setVariable [ "levels", count (getArray (configFile >> "SyndikatGear" >> "primaryWeapons")) ];
  _logic setVariable [ "actualLevel", 0 ];


	SVAR(_logic, "uiToUpdate", []);

	private _UIeh = addMissionEventHandler
  ["Map",
   {
		 scriptName "_Init/MapHandler";
    params ["_mapIsOpened", "_mapIsForced"];
   	if(BIS_SyndikatPool_Debug && {_mapIsOpened}) then
	 	{
		 	USVAR("_syndikatAgentDebugUI", findDisplay 46 createDisplay "RscSyndikatPool");
			["UpdateDebugInfo", [MGVAR("BIS_OM_SyndikatAgent_logic")] ] call BIS_fnc_OM_ModuleSyndikatAgent;
			[] spawn
			{
				scriptName "_Init/MapHandler/Spawn1";
				private _controls = [IDC_SYNDIKATPOOL_EDITACTUALPOOLVALUE, IDC_SYNDIKATPOOL_EDITPOOLMINVALUE, IDC_SYNDIKATPOOL_EDITPOOLMAXVALUE, IDC_SYNDIKATPOOL_EDITINCOMEUNITSVALUE, IDC_SYNDIKATPOOL_EDITINCOMETIMEVALUE];
				{
					ctrlEnable [_x, false];
				} foreach _controls;
				uiSleep 2;
				{
					ctrlEnable [_x, true];
				} foreach _controls;
			};
		}
		else
		{
			(UGVAR("_syndikatAgentDebugUI")) closeDisplay 1;
		};
   }
  ];

  [_logic] spawn
  {
    scriptName "syndikatAgent_eventLoop";
    //timer 1 - for selecting events
    private _logic = _this select 0;
    private "_diff";
    while {true} do
    {
      private _originalDate = date;
      private _originalPeriod = GVAR(_logic, "EventSelectionPeriod");
      private _newDate = [_originalDate, _originalPeriod] call BIS_fnc_CalculateDateTime;
      WaitUntil
      {
        Sleep 1;
        if(_originalPeriod != GVAR(_logic, "EventSelectionPeriod") || _logic getVariable ["BIS_restartTimer", false]) then
        {
          _originalDate = date;
          _originalPeriod = GVAR(_logic, "EventSelectionPeriod");
          _newDate = [_originalDate, GVAR(_logic, "EventSelectionPeriod")] call BIS_fnc_CalculateDateTime;
          _logic setVariable ["BIS_restartTimer", false];
        }; //value updated, reset the datetime the new event will be computed from

        /* ([date, _newDate] call BIS_fnc_IsDateTimeNewer)
        ||
        (([date, _newDate] call BIS_fnc_CompareDateTimes) isEqualType true) */
         _diff = [date, _newDate, "m"] call BIS_fnc_CalculateDateTime;
         _diff <= 0;
      };
      ["EventSelectTarget", [_logic, _diff]] spawn BIS_fnc_OM_ModuleSyndikatAgent
    };
  };

  [_logic] spawn
  {
    scriptName "syndikatAgent_teamIncrease";
    //timer 2 - for increasing the team
    private _logic = _this select 0;

    while {true} do
    {
      private _originalDate = date;
      private _poolIncomeUnits = GVAR(_logic, "PoolIncomeUnits");
      private _poolIncomeTime = GVAR(_logic, "poolIncomeTime");
      private _newDate = _originalDate;


      WaitUntil
      {
        Sleep 1;
        if( _poolIncomeUnits != GVAR(_logic, "PoolIncomeUnits")) then {_poolIncomeUnits = GVAR(_logic, "PoolIncomeUnits"); _originalDate = date; "income units changed" call BIS_fnc_log; };
        if( _poolIncomeTime != GVAR(_logic, "poolIncomeTime")) then {_poolIncomeTime = GVAR(_logic, "poolIncomeTime"); _originalDate = date; ["income TIME changed to: %1", _poolIncomeTime] call BIS_fnc_log;}; //in case the value is changed during the game. It would add many units in case there was a lower value before.
        _newDate = [_originalDate, GVAR(_logic, "poolIncomeTime")] call BIS_fnc_CalculateDateTime;
        ([date, _newDate] call BIS_fnc_IsDateTimeNewer)     //date is greater
        ||
        (([date, _newDate] call BIS_fnc_CompareDateTimes) isEqualType true)
      };
      private _unitsToAdd = (round ((([ _originalDate, date ] call BIS_fnc_CalculateDatetime))/_poolIncomeTime)) * _poolIncomeUnits;
      /* ["Wanted Units: %1", _unitsToAdd] call BIS_fnc_LogFormat; */
      [
        "AddUnitsToPool",
        [ [nil, _unitsToAdd] ]
      ] spawn BIS_fnc_OM_ModuleSyndikatAgent;
      //hack - should be part of the Syndikat Team
      if(SYNDIKATTEAM_IS_FLAG_ENABLED(DIRECT_ACTIVATION_ENABLED)) then
      {
        switch (GET_REPUTATION_LEVEL()) do
        {
            case (0): { [ MODE_ADD_UNITS_FROM_DATA, [ [ [nil,round random[1,1,2] ] ]] ] call BIS_fnc_OM_ModuleSyndikatTeam };
            case (1): { [ MODE_ADD_UNITS_FROM_DATA, [ [ [nil,round random[2,3,4] ] ]] ] call BIS_fnc_OM_ModuleSyndikatTeam };
            case (2): { [ MODE_ADD_UNITS_FROM_DATA, [ [ [nil,round random[3,3,5] ] ]] ] call BIS_fnc_OM_ModuleSyndikatTeam };
        };
      };

    };
  };

  [_logic] spawn
  {
    scriptName "syndikatAgent_debugUpdate";
    //timer 3 - for debug update
    while {true} do
    {
      ["UpdateDebugInfo", [_this select 0]] spawn BIS_fnc_OM_ModuleSyndikatAgent;
      Sleep SYNDIKATPOOL_DEBUG_UPDATE_INTERVAL;
    };
  };
};

private _AddUnitsToPool =
{
	scriptName "_AddUnitsToPool";
  params [ "_logic", ["_unitsArray", [ ["SyndikatSoldier_Rifleman",1] ], [ [[]] ] ] ];
  private _poolNumber = [GVAR( _logic, "basicGroup" )] call BIS_fnc_OM_getNumberFromData;
  private _poolMax = GVAR(_logic, "PoolMax");
  private _level = _logic getVariable "actualLevel";
  {
    private _unitType = _x#0;
    private _selectRandom = if(isNil "_unitType") then {true} else {false};
    private _unitsToAdd = (_poolMax - _poolNumber) min _x#1;                    //either the wanted nr of units or just the rest to the poolMax

    for "_iLoop" from 1 to _unitsToAdd do
    {
      if( _selectRandom) then
      {
        _unitType = SOLDIER_CLASSES selectRandomWeighted SOLDIER_WEIGHTS;       //if the class is not defined then select random class (weighted)
      };

      private _classes = (GET_CONFIG_ENTRY( "types", _unitType ));
      if( isNull(_classes) ) exitWith { false };                                //this unit class doesn't exist
      private _selectedClass = selectRandom (getArray _classes);                 //select a class from the array
      private _unitSkill = [_level] call BIS_fnc_OM_ModuleSyndikatTeam_getSkill;
      private _unitLoadout = [_level, _selectedClass] call BIS_fnc_OM_ModuleSyndikatAgent_setGear;
      private _unitPosition = selectRandom (getArray (GET_CONFIG_ENTRY("positions",_unitType))); //TODO: function get free position (post)
      private _newUnit = [_selectedClass, "", _unitPosition, 0, _unitSkill, 0, "AUTO", _unitLoadout, "", false, []];
      private _newGroup = [independent, GROUP_BEHAVIOR, [], [_newUnit]];
      ["SYNDIKATAGENT: adding %1 to pool on level %2", _selectedClass, _level] call BIS_fnc_logFormat;
      SYNDIKATPOOL_GROUPS pushBack _newGroup;
      _poolNumber = _poolNumber + 1;
      SVAR( _logic, "PoolPower", [GVAR(_logic, "basicGroup")] call BIS_fnc_OM_ModuleSyndikatAgent_GetTeamPower);
  		SVAR(_logic, "PoolInitNumber", _poolNumber);
      [IDC_SYNDIKATPOOL_TEXTACTUALPOOLVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
      [IDC_SYNDIKATPOOL_TEXTBASEPOWERVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
  } forEach _unitsArray;
};

private _EventSelectTarget =
{
		scriptname "_EventSelectTarget";
    params [ ["_logic", objNull, [objNull] ], "_diff"];
    ["SYNDIKATAGENT: Starting to select the target on %1", date] call BIS_fnc_logFormat;
    ["SYNDIKATAGENT: the actual pool is %1", GVAR(_logic, "PoolInitNumber")] call BIS_fnc_logFormat;
    if (is3DEN) exitWith {};
    if ( ([] call BIS_fnc_OM_ModuleSyndikatAgent_GetActiveQuests) >= (GVAR(_logic, "MaximumActiveSupports")) ) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: there are already %1 active Syndikats assaults", GVAR(_logic, "MaximumActiveSupports")] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "Quests already active" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
    if( _diff <= (-1*ALLOWED_DELAY)) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: Allowed delay < %1", ALLOWED_DELAY] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "Allowed delay" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };

    private _lastEventDate = GVAR( _logic, "LastEventDate");
    private _minDate = [_lastEventDate, MIN_TIME_AFTER_EVENT_FINISHED] call BIS_fnc_CalculateDateTime;
    if!( [date, _minDate] call BIS_fnc_IsDateTimeNewer ) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: Too early, the last fight was on %1", _lastEventDate] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "Too early after Event" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };

    //private _probabilityOfEvent = linearConversion [0, TEMP_MAX_BASES_ON_ISLAND, count _sectors, 0,1,true];
    if!( random 1 <= (_logic getVariable ["SupportProbability", 1]) ) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: Probability to small", _lastEventDate] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "Probability" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
    if( GVAR( _logic, "poolInitNumber" ) <  GVAR( _logic, "poolMin" ) ) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: Number of Syndikats is under minimum: %1",  GVAR( _logic, "poolMin" )] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "Low Pool" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
  	private _sectors = [] call BIS_fnc_OM_ModuleSyndikatAgent_UpdateSectorsData;
    ["SYNDIKATAGENT: Target selection. %1 sectors before filter",  count _sectors] call BIS_fnc_logFormat;
    [] spawn _DumpSectorsInfo;
		private _filteredSectors =
		[
			_sectors,
			GVAR( _logic, "poolPower" ),
			GVAR( _logic, "minTimeBetweenEvents" ),
      IN_DEFAULT_RANGE_FROM_PLAYER
		] call BIS_fnc_OM_ModuleSyndikatAgent_FilterSectors;
    ["SYNDIKATAGENT: Target selection. %1 filtered sectors",  count _filteredSectors] call BIS_fnc_logFormat;
    [_filteredSectors] spawn _DumpSectorsInfo;

    if( count _filteredSectors == 0 ) exitWith
    {
      ["SYNDIKATAGENT: Target selection FAILED: there's 0 sectors available", 0] call BIS_fnc_logFormat;
      ["SYNDIKATAGENT: Filter params: poolPower must be: %1, pause between assaults: %2", GVAR( _logic, "poolPower" ), GVAR( _logic, "minTimeBetweenEvents" )] call BIS_fnc_logFormat;
      SVAR( _logic, "baseObject", "0 sectors available" );
      [IDC_SYNDIKATPOOL_TEXTBASESELECTEDVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
		private _randomIndex = floor random (count _filteredSectors);
		private _baseSelected =  _filteredSectors select _randomIndex;
		SVAR( _logic, "baseObject", _baseSelected#BASE_NAME );

      )];
      private _basicGroup = GVAR( _logic, "basicGroup");

      [IDC_SYNDIKATPOOL_TEXTTEAMPOWERVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;

      private _newDate = [date, (GVAR(_logic,"EventSelectionPeriod"))] call BIS_fnc_CalculateDateTime;

      [IDC_SYNDIKATPOOL_TEXTNEXTEVENTDATEVALUE] call BIS_fnc_OM_ModuleSyndikatAgent_EnqueueDebugInfo;
    };
    [ MODE_ADDTOQUEUE, [ [ _codeForQueue, [_logic, _copyOfUnits, _baseSelected], 1, 120, {"Quest created from queue" call BIS_fnc_Log},nil,nil,nil,nil, "SyndikatAgent"] ] ] call BIS_fnc_OM_moduleActionQueue;

  {
      params[ ["_tabs", 1, [99] ] ];
