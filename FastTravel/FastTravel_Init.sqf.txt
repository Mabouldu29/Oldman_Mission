BIS_OM_FastTravel_condition =
{
	scriptName "BIS_OM_FastTravel_condition";
	private _ret = false;
	private _index = BIS_activeSectors findIf { _x distance player <= MAX_DISTANCE_FROM_SECTOR && { !(_x getVariable ["BIS_OM_fTravelPoints", []] isEqualTo []) } };
	if (_index != -1) then
	{
		private _sector = BIS_activeSectors select _index;
		BIS_FASTTRAVEL_LOGIC setVariable ["BIS_OM_playersSector", _sector];
		_ret = ((_sector getVariable "BIS_OM_fTravelPoints") select { call compile ( _x select 2 ) } findIf { _x select 0 distance player < MAX_DISTANCE_FROM_TRAVELPOS }) != -1;
	};
	_ret;
};
