BIS_addEH_takeWeapShed =
{
    OM_EH_take_weaponshed = player addEventHandler ["Take",
    {
        if ((_this select 1)  in (BIS_OM_ShedWeaponWall getVariable "BIS_wallWeaponHolders")) exitWith
        {
            _order = 0;
            BIS_holder = (_this select 1);
            {
                if (_x isEqualTo (_this select 1)) exitwith {_order = _forEachIndex};
            } forEach BIS_wallWeaponHolders;
            BIS_holder attachTo [
                BIS_OM_ShedWeaponWall,
                (BIS_OM_ShedWeaponWall getVariable "offsets") select _order
            ];
            detach BIS_holder;
            BIS_wallWeaponHolders set [_order, BIS_holder];
            BIS_OM_ShedWeaponWall setVariable ["BIS_wallWeaponHolders", BIS_wallWeaponHolders, true];
            closeDialog 0;
        };
    }];
};

BIS_fnc_OM_putWeapon =
{
    params ["_unit", "_weapon", "_position", ["_whType", "GroundWeaponHolder"]];
    _unit addEventHandler ["Put",
    {
        _this select 0 removeEventHandler ["Put", _thisEventHandler];
        clearItemCargo (_this select 1);
    }];
    isNil
    {
        BIS_holder = _whType createVehicle getpos player;
        BIS_holder addItemCargo ["None", 1];
        BIS_wallWeaponHolders = BIS_OM_ShedWeaponWall getVariable "BIS_wallWeaponHolders";
        {
            if (isNull _x) exitWith
            {
                BIS_holder attachTo [
                    BIS_OM_ShedWeaponWall,
                    (BIS_OM_ShedWeaponWall getVariable "offsets") select _forEachIndex
                ];
                detach BIS_holder;

                BIS_wallWeaponHolders set [_forEachIndex, BIS_holder];
                BIS_OM_ShedWeaponWall setVariable ["BIS_wallWeaponHolders", BIS_wallWeaponHolders, true];
                //closeDialog 0;
            };
        } forEach BIS_wallWeaponHolders;
        _unit action ["PutWeapon", BIS_holder, _weapon];
    };
};
