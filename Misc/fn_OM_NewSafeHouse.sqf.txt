#define PRICE_BASIC             99
#include "\a3\Missions_F_Oldman\Systems\defines\commonDefines.inc"
//TODO: refresh displayed price in the HoldAction in case the price is changed


params ["_bed", "_free", "_owner"];
//--- in case player has good relation ship with syndicat he doesnt need to pay at all
//if (true) then {_free=0};
private _actionID = -1;
switch (_free) do
{
    case 0:
    {
        // FOR FREE
        // {
        //--- store variable that player doesnt need to pay for rest place
        player setVariable ["paidSavePoint",false];
        _actionID = [
            _bed,
            Localize "STR_A3_OM_HoldAction_NewSafeHouseSleep",
            "\a3\Ui_F_Oldman\Data\IGUI\Cfg\HoldActions\holdAction_sleep2_ca.paa",
            "\a3\Ui_F_Oldman\Data\IGUI\Cfg\HoldActions\holdAction_sleep2_ca.paa",
            "player distance _target < 3 && !(player getvariable ['BIS_IsSleeping',false])",
            "player distance _target < 3",
            {
            //--- when hold action starts
            },
            {
            //--- tick every frame (out of 24)
            },
            {
            //--- when holdaction ends
            [_target] spawn
            {
                //BIS_TimeToSkipWhileSleeping = 1/3;
                player setvariable ["BIS_IsSleeping",true];
                sleep 0.5;
                player setVariable ["AtGasStation",true];
                [ (_this select 0) getVariable ["forcedSleepTime",-1] ] execVM "\a3\Missions_F_Oldman\Systems\UI\Sleeping\Createrestui.SQF";
                waitUntil {!isnull findDisplay IDD_REST_DISPLAY};
                waituntil {isnull findDisplay IDD_REST_DISPLAY};
                //createdialog "RscTestControlTypes";
                //waituntil {isnull findDisplay 999};
                if (!isNil "BIS_SleepTimeSelected" && {BIS_SleepTimeSelected}) then
                {
                    player setvariable ["BIS_IsSleeping",true];
                } else {
                    player setvariable ["BIS_IsSleeping",false];
                    BIS_SleepingTime = [nil,0]
                };
                BIS_SleepTimeSelected = false;
                player setVariable ["AtGasStation",false];
            };
            },
            {
            //--- when action is terminated
            },
            [],
            1,
            nil,
            false,
            false
        ] call BIS_fnc_holdActionAdd;
    };
    case 1:
    {
        // PAID
        //{
        //--- store variable that player have to pay for rest place
        player setVariable ["paidSavePoint",true];
        if((_bed getVariable "BIS_Owner") == CIVILIAN) then
        {
          missionNamespace setVariable ["BedPrice", missionNamespace getVariable ["BedPriceCivil",PRICE_BASIC]]
        }
        else
        {
          missionNamespace setVariable ["BedPrice", missionNamespace getVariable ["BedPriceSyndikat",PRICE_BASIC]]
        };
        _actionID = [
        _bed,
        format [ localize "STR_A3_OM_fn_OM_NewSafeHouse.sqf0",(missionNamespace getVariable ["BedPrice", PRICE_BASIC])],
				"\a3\Ui_F_Oldman\Data\IGUI\Cfg\HoldActions\holdAction_sleep2_ca.paa",
				"\a3\Ui_F_Oldman\Data\IGUI\Cfg\HoldActions\holdAction_sleep2_ca.paa",
        "player distance _target < 3 && !(player getvariable ['BIS_IsSleeping',false])",
        "player distance _target < 3 && ([3] call BIS_fnc_OM_ModuleEconomy >= (missionNamespace getVariable 'BedPrice'))",
        {
        //--- when hold action starts
        },
        {
        //--- tick every frame (out of 24)
        },
        {
        //--- when holdaction ends
        //[ 10, [(missionNamespace getVariable "BedPrice")]] call BIS_fnc_OM_ModuleEconomy;
        [] spawn
        {
            //BIS_TimeToSkipWhileSleeping = 1/3;
            player setvariable ["BIS_IsSleeping",true];
            sleep 0.5;
            player setVariable ["AtGasStation",true];
            [] execVM "\a3\Missions_F_Oldman\Systems\UI\Sleeping\Createrestui.SQF";
            waitUntil {!isnull findDisplay IDD_REST_DISPLAY};
            waituntil {isnull findDisplay IDD_REST_DISPLAY};
            //createdialog "RscTestControlTypes";
            //waituntil {isnull findDisplay 999};
            if (BIS_SleepTimeSelected) then
            {
                player setvariable ["BIS_IsSleeping",true];
                BIS_SleepTimeSelected = false
            } else {
                player setvariable ["BIS_IsSleeping",false];
                BIS_SleepTimeSelected = false;
                BIS_SleepingTime = [nil,0]
            };
            player setVariable ["AtGasStation",false];
        };
        },
        {
        //--- when action is terminated
        },
        [],
        1,
        nil,
        false,
        false
        ] call BIS_fnc_holdActionAdd;
    };
    default {};
};

_actionID
