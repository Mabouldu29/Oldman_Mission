BIS_fnc_OM_showPrice =
{
    _mousePos = getMousePosition;
    params ["_itemText", "_itemClass", "_container"];
    if(!isNil "_itemClass") then
    {
        //--- get the price
        private _price = [ MODE_GET_EQUIP_PRICE,[ _itemClass, if( _container in PLAYER_INVENTORY_CONTAINERS) then {true} else {false} ] ] call BIS_fnc_OM_ModuleEconomy;
        //hint format ["%1\n%2\n%3",_itemText,_itemClass,_container];
        if (_price isEqualType 0) then {
            //--- set default color
            private _color = '#cc3300';
            if (_price < ([3] call BIS_fnc_OM_ModuleEconomy)) then {_color = '#FFA500'};
            //--- change to green if is it in player inventory (uniform,vest,backpack)
            if (_container in PLAYER_INVENTORY_CONTAINERS) then {_color = '#61af64'};
            //--- show tooltip

            [_price,_color,_mousePos] spawn BIS_fnc_OM_priceTooltip;
        };
    };
};

BIS_fnc_RemovePrefix =
{
    #define PREFIXES ["ITEM", "WEAPON", "MAGAZINE"]
    params["_string"];

    private _array = _string splitString "_";
    if(toUpper(_array select 0) in PREFIXES) then
    {
        _array deleteAt 0;
        _array joinString "_"
    }
    else {_string};
};
