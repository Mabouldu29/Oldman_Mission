{
	params ["_carToExplode"];

	while {alive _carToExplode} do
	{
		waituntil {visiblemap};
		if !(alive _carToExplode) exitWith {};
		_carToExplode setVariable ["marker", [MODE_CREATE, ["SmartMarker_T_CarChargeActive_F", format ["BIS_mrk_car_%1", vectorMagnitude (getpos _carToExplode) * 10], getPos _carToExplode, false, nil, false]] call BIS_fnc_OM_moduleSmartMarkers];

		[MODE_DELETE, [_carToExplode getVariable "marker"]] call BIS_fnc_OM_moduleSmartMarkers;
	};
	[MODE_DELETE, [_carToExplode getVariable "marker"]] call BIS_fnc_OM_moduleSmartMarkers;
