private _fnc_quest_getCfgData = {
	params["_cfg", "_property", ["_quest",objNull], ["_isBool", false], ["_defaultType", ""] ];
	private _questClassName = "cfgQuests";
	
	private _defaultType = "";
	if (_property isEqualType []) then {
		_defaultType = _property#1;
		_property =	_property#0;
	};
	
	private _cfgDataValue = ([_questClassName, _cfg, _property] call BIS_fnc_getCfgData);
	if (isNil "_cfgDataValue") then {
		_cfgDataValue = objNull;
	} else {
		if ((!(_defaultType isEqualTo "")) && (_cfgDataValue isEqualTo _defaultType)) then {
			_cfgDataValue = [_cfg, _defaultType, _quest, _isBool] call _fnc_quest_getCfgData;
		};
	};
	if (_isBool) then {
		_cfgDataValue = call (compile _cfgDataValue);
	};
	
	if (!(isNull _quest)) then {
		 SVAR(_quest, _property, _cfgDataValue);
	};
	_cfgDataValue
};
