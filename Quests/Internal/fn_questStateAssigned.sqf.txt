	if (_useSmartMarker) then {
		private _smartMarkerName = format["SmartMarker_%1_%2_F",GVAR(_quest,"smartMarkerPrefix"), ([_iconType] call bis_fnc_capitalizeString)];
		private _smartMarkerTitle = _title;
		if (!(_startTimeText isEqualTo "")) then {
			_smartMarkerTitle = format["%1 (%2)", _smartMarkerTitle, _startTimeText];
		};
		//["quest CREATE SmartMarker %1 %2", _smartMarkerName, _startTimeText] call bis_fnc_logFormat;
		private _markerParams = [ [SM_TOOLTIP_TITLE, _smartMarkerTitle],[SM_TOOLTIP_DESCR, GVAR(_quest, "markerDescription")] ];
		private _markerPicture = GVAR(_quest, "markerPicture");
		if (!(_markerPicture isEqualTo "")) then {
			_markerParams pushBack ([SM_TOOLTIP_IMAGE, _markerPicture]);
		};
		//CREATE MARKER WITH CUSTOM DATA                (use macros: SM_NAME, SM_TOOLTIP_TITLE, SM_TOOLTIP_DESCR, SM_TOOLTIP_IMAGE)
		private _showSmartMarker = GVAR(_quest, "showSmartMarker");
		private _showSmartMarkerFlag = if (_showSmartMarker) then { MODE_SHOW } else { MODE_HIDE };
		_markerID = [MODE_CREATE, [_smartMarkerName, _currentTemplateName, _questPos, !_showSmartMarker , _markerParams]] call BIS_fnc_OM_moduleSmartMarkers;

		private _markerData = GVAR(_quest, "markerData");
		_markerData params["_markerTitle", "_markerTooltipTitle", "_markerTooltipText"];

		if (_markerTitle != "") then { _paramsMarker pushBack [SM_NAME, _markerTitle]; };
		if (_markerTooltipTitle != "") then { _paramsMarker pushBack [SM_TOOLTIP_TITLE, _markerTooltipTitle]; };
		if (_markerTooltipText != "") then { _paramsMarker pushBack [SM_TOOLTIP_DESCR, _markerTooltipText]; };
		if (!(_paramsMarker isEqualTo [])) then { [MODE_CHANGEATTRIBUTES, [_markerID, _paramsMarker ] ] call BIS_fnc_OM_moduleSmartMarkers; };
	} else {
		_taskPos = _questObject;
	};
