#include "\A3\Missions_F_Oldman\Systems\defines\questDefines.inc";

_getQuestSpecificFunctions = {
	params["_questType"];
	
	private _functionNames =
	[
		"questInit",
		"questProgress",
		"questFinished"
	];
	_path =  format["\A3\Missions_F_Oldman\Systems\scripts\Quests\Internal\%1\",_questType];
	[
		_path,
		"bis_fnc_" + _questType,
		_functionNames
	] call bis_fnc_loadFunctions;
};

private _fnc_quest_initModule = {
	params["_module"];

	private _def = [0,0];
	private _area = GVARS(_module, "objectArea", _def);
	private _areaSize = (_area#0) max (_area#1);
	private _templateName =  GVAR(_module,"TaskTemplate");
	private _questType =  [configfile >> "cfgVehicles" >> typeOf _module >> "questType"] call BIS_fnc_getCfgData;
	private _areaPos = getPos _module;
	[_areaPos, _areaSize, _templateName, _questType]
};

private _fnc_getAreaObjects = {
	params["_areaPos", "_areaSize", "_className"];
	(_areaPos nearObjects ["All", _areaSize] select {if (_x isKindOf "Man") then { _x canAddItemToBackpack "Default"} else {_x canAdd "Default"}} )
};
