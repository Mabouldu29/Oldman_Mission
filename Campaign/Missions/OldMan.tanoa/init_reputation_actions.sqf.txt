/*
	==============================================================================
	HERE DEFINE THE ACTIONS WHICH SHOULD BE EXECUTED ON REACHING A CERTAIN LEVEL
	==============================================================================
*/

private _level3 =
{
	params [ ["_actualLevel", 3, [999] ], ["_prevLevel", 3, [999] ] ];
	ECONOMY_SET_ITEM_EXTRACHARGE_COST(-35);
	ECONOMY_SET_ITEM_EXTRACHARGE_PURCHASE(35);
	[_actualLevel,_prevLevel] spawn BIS_fnc_OM_repNotification;
	missionNamespace setvariable ["BedPriceSyndikat", PRICE_BASIC/_actualLevel];
};
private _level2 =
{
	params [ ["_actualLevel", 2, [999] ], ["_prevLevel", 2, [999] ] ];
	ECONOMY_SET_ITEM_EXTRACHARGE_COST(-20);
	ECONOMY_SET_ITEM_EXTRACHARGE_PURCHASE(20);
	[_actualLevel,_prevLevel] spawn BIS_fnc_OM_repNotification;
	missionNamespace setvariable ["BedPriceSyndikat", PRICE_BASIC/_actualLevel];
	missionNameSpace setVariable ["BIS_SynikatSmugglersRestricted", false];
};
private _level1 =
{
	params [ ["_actualLevel", 1, [999] ], ["_prevLevel", 1, [999] ] ];
	ECONOMY_SET_ITEM_EXTRACHARGE_COST(-10);
	ECONOMY_SET_ITEM_EXTRACHARGE_PURCHASE(10);
	SUPPORT_SET_EVENT_PROBABILITY(1);
	SYNDIKATTEAM_SET_FLAG( INDIRECT_ACTIVATION_ENABLED );
	SYNDIKATTEAM_SET_FLAG( DIRECT_ACTIVATION_ENABLED );
	if(isNil "BIS_infoSyndikat_SyndikatTeam") then
	{
		BIS_infoSyndikat_SyndikatTeam = player createDiaryRecord ["SyndikatInfo", [  localize "STR_A3_init_reputation_actions.sqf0", [] call BIS_fnc_generateSyndikatsInfoText_SyndikatTeam], taskNull, "", false];
	};
	//Syndikat Team can be called over the radio
	//removed only if the reputation gets bad (Boss reacts over the phone related to the reputation)
	if(isNil {missionNameSpace getVariable "SyndikatTeamRequestCallID"}) then
	{
		missionNameSpace setVariable
		[
			"SyndikatTeamRequestCallID",
			["Boss", localize "STR_A3_init_reputation_actions.sqf1", {[3,[true]] call BIS_fnc_OM_moduleSyndikatTeam}, true, false] call BIS_fnc_OM_phone_addCall	//TODO: localize
		]
	};

	[_actualLevel,_prevLevel] spawn BIS_fnc_OM_repNotification;
	missionNamespace setvariable ["BedPriceSyndikat", PRICE_BASIC/_actualLevel];
};
private _level0 =
{
	params [ ["_actualLevel", 0, [999] ], ["_prevLevel", 0, [999] ] ];
	missionNameSpace setVariable ["BIS_mainCampRestricted", false];
	missionNameSpace setVariable ["BIS_TempleCampRestricted", false];
	missionNameSpace setVariable ["BIS_AllCampsRestricted", false];
	missionNameSpace setVariable ["BIS_SyndikatRestpointsRestricted", false];
	ECONOMY_SET_ITEM_EXTRACHARGE_COST(0);
	ECONOMY_SET_ITEM_EXTRACHARGE_PURCHASE(0);
	SUPPORT_SET_EVENT_PROBABILITY(0.65);
	if (!isNil "_firstTimeRepChanged") then {SYNDIKATTEAM_SET_FLAG( DIRECT_ACTIVATION_ENABLED )};
	if(isNil "BIS_infoSyndikat_SyndikatAssault") then
	{
		BIS_infoSyndikat_SyndikatAssault = player createDiaryRecord ["SyndikatInfo", [  localize "STR_A3_init_reputation_actions.sqf2", [] call BIS_fnc_generateSyndikatsInfoText_SyndikatAssault], taskNull, "", false];
	};
	if(isNil "BIS_infoSyndikat_DropOffPoint") then
	{
		BIS_infoSyndikat_DropOffPoint = player createDiaryRecord ["SyndikatInfo", [  localize "STR_A3_init_reputation_actions.sqf3", [] call BIS_fnc_generateSyndikatsInfoText_DropOffPoints], taskNull, "", false]
	};
	if (!isNil "_firstTimeRepChanged") then
	{
		[_actualLevel,_prevLevel] spawn BIS_fnc_OM_repNotification;
	}
	else
	{
		private _firstTimeRepChanged = true;
		["Syndikatinfo", true] call BIS_fnc_OM_phone_enableButton; //---enable Syndikat Info button
	};
	missionNamespace setvariable ["BedPriceSyndikat", PRICE_BASIC];
	if(isNil "BIS_infoSyndikat_safehouse") then
	{
		BIS_infoSyndikat_safehouse = player createDiaryRecord ["SyndikatInfo", [  localize "STR_A3_init_reputation_actions.sqf4", [] call BIS_fnc_generateSyndikatsInfoText_safehouse], taskNull, "", false];
	};
};
private _levelMinus1 =
{
	params [ ["_actualLevel", -1, [999] ], ["_prevLevel", -1, [999] ] ];
	if(_actualLevel != _prevLevel) then{ [_actualLevel,_prevLevel] spawn BIS_fnc_OM_repNotification; };		//mission started, do not activate the speech
	missionNameSpace setVariable ["BIS_mainCampRestricted", (_actualLevel == _prevLevel)];									//TODO: remove if not needed anymore
	missionNameSpace setVariable ["BIS_TempleCampRestricted", true];
	missionNameSpace setVariable ["BIS_AllCampsRestricted", true];
	missionNameSpace setVariable ["BIS_SyndikatRestpointsRestricted", true];
	missionNameSpace setVariable ["BIS_SynikatSmugglersRestricted", true];
	ECONOMY_SET_ITEM_EXTRACHARGE_COST(10);
	ECONOMY_SET_ITEM_EXTRACHARGE_PURCHASE(-10);
	SUPPORT_SET_EVENT_PROBABILITY(0);
	SYNDIKATTEAM_CLEAR_FLAG( DIRECT_ACTIVATION_ENABLED + INDIRECT_ACTIVATION_ENABLED );		//Syndikat Team disabled (neither by direct contact in a camp nor over the radio (TODO))
	//Remove the request call from the diary menu
	if!(isNil {missionNameSpace getVariable "SyndikatTeamRequestCallID"}) then
	{
		[missionNameSpace getVariable "SyndikatTeamRequestCallID"] call BIS_fnc_OM_phone_removeCall;
		missionNameSpace setVariable ["SyndikatTeamRequestCallID", nil];
	};

	missionNamespace setvariable ["BedPriceSyndikat", PRICE_BASIC*10000];
};
