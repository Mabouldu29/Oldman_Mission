// --- cancel irrelevant tasks - loop check

[] spawn {
	_allQuestClasses = "TRUE" configClasses (missionConfigFile >> "CfgQuests");
	_allCancelConditions = _allQuestClasses apply {getText (_x >> "cancelCondition")};
	_allTitles = _allQuestClasses apply {getText (_x >> "title")};
	{
		if (_x != "") then {
			_allCancelConditions set [_forEachIndex, compile _x];
		} else {
			_allCancelConditions set [_forEachIndex, {FALSE}];
		};
	} forEach _allCancelConditions;
	while {TRUE} do {
		_allPlayerTasks = [player] call BIS_fnc_tasksUnit;
		_allPlayerTasksActive = _allPlayerTasks select {!([_x] call BIS_fnc_taskCompleted)};
		_allPlayerTasksActiveTitles = _allPlayerTasksActive apply {(([_x] call BIS_fnc_taskDescription) # 1) # 0};
		{
			_title = _x;
			_questID = _allTitles find _title;
			_assignedQuests = [["questState", 1],"", false, nil, true] call bis_fnc_questGetData;
			_assignedQuestTitles = _assignedQuests apply {_x getVariable "title"};
			_questActualID = _assignedQuestTitles find _title;
			if (_questActualID != -1) then {
				_questActual = _assignedQuests # _questActualID;
				if (_questID != -1) then {
					_cancelCondition = _allCancelConditions # _questID;
					if (call _cancelCondition) then {
						_parentTask = _allPlayerTasksActive # _forEachIndex;
						[_questActual getVariable "questId", 6] call bis_fnc_questSetState;
						[1, [GVAR(_questActual, "questTemplate")]] call BIS_fnc_OM_moduleSmartMarkers;
						{
							_title = (([_x] call BIS_fnc_taskDescription) # 1) # 0;
							_questActualID = _assignedQuestTitles find _title;
							if (_questActualID != -1) then {
								_questActual = _assignedQuests # _questActualID;
								[1, [GVAR(_questActual, "questTemplate")]] call BIS_fnc_OM_moduleSmartMarkers;
							};
						} forEach ([_parentTask] call BIS_fnc_taskChildren);
					};
				};
			};
			sleep 0.1;
		} forEach _allPlayerTasksActiveTitles;
		sleep 5;
	};
};
