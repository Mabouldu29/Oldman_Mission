private _cHandleDropPointBoxes = format
["
	[_this] spawn
	{
		scriptName '_cHandleDropPointBoxes';
		private _boxObject = _this select 0;
		waitUntil {sleep 1; call {%2} && !isNull _boxObject};
		if (isNull (uiNamespace getVariable ['BIS_OM_gearTransfer', controlNull])) then
		{
			uiNamespace setVariable ['BIS_OM_gearTransfer', (findDisplay 46) ctrlCreate ['RscStructuredText', 9999014]];
		};
		_gearTransfer = uiNamespace getVariable 'BIS_OM_gearTransfer';
		_gearTransfer ctrlShow FALSE;
		_gearTransfer ctrlSetBackgroundColor [0,0,0,0];
		_gearTransfer ctrlCommit 0;
		private _contOpenHndl = _boxObject addEventHandler
		[
			'ContainerOpened',
			{
				if( !call {%2} ) exitWith {};
				(_this select 0) spawn
				{
					scriptName 'moduleDepot/ContainerOpenedEventhandler';
					waitUntil {!isNull (findDisplay 602)};
					if (isNull (uiNamespace getVariable ['BIS_OM_gearTransfer', controlNull])) then
					{
						uiNamespace setVariable ['BIS_OM_gearTransfer', (findDisplay 46) ctrlCreate ['RscStructuredText', 9999014]];
					};
					_gearTransfer = uiNamespace getVariable 'BIS_OM_gearTransfer';
					_gearTransfer ctrlSetPosition ctrlPosition (findDisplay 602 displayCtrl 6307);
					_gearTransfer ctrlCommit 0;
					_gearTransfer ctrlShow TRUE;
					_gearTransfer ctrlSetStructuredText parseText ('<t align = ''left''>' + (localize 'STR_A3_moduleDepot_close') + '</t><t align = ''right'' color = ''#61af64''>$0</t>');
					_totalItemsLoaded = [];
					while {ctrlShown _gearTransfer} do
					{
						waitUntil {sleep 0.5; !(_totalItemsLoaded isEqualTo [itemCargo _this, weaponCargo _this, magazineCargo _this, backpackCargo _this])};
						_totalItemsLoaded = +[itemCargo _this, weaponCargo _this, magazineCargo _this, backpackCargo _this];
						_gearTransfer ctrlSetStructuredText parseText ('<t align = ''left''>' + (localize 'STR_A3_moduleDepot_close') + '</t><t align = ''right'' color = ''#61af64''>' + (str([_this, FALSE] call BIS_fnc_OM_sellEquipment)) + '</t>');
					};
				};
			}
		];
		private _contCloseHndl = _boxObject addEventHandler
		['ContainerClosed',
			{
				if( !call {%2} ) exitWith {};
				(uiNamespace getVariable ['BIS_OM_gearTransfer', controlNull]) ctrlShow FALSE;
				[(_this select 0), TRUE] spawn BIS_fnc_OM_sellEquipment;
			}
		];
		waitUntil {sleep 1; isNull _boxObject};
		_boxObject removeEventHandler ['ContainerOpened', _contOpenHndl];
		_boxObject removeEventHandler ['ContainerClosed', _contCloseHndl];
	};
	", "%1", _cond
];
