[] spawn {
	scriptName "qrf_moduleAgent";
	private _timeoutDelay = QRF_TIMEOUT_CHECK_DELAY;
	/* private _timeoutDelay = 1; */
	while {alive player} do {
		for "_i" from (count (["lastCooldownTime"] call bis_fnc_qrf_areaParam) -1) to 0 step -1 do {
			PROFILE_INFO("qrf_moduleAgentSpawn1","qrf_moduleAgentSpawn1",1);
			if ( !(["elapsed", _i, nil, false] call bis_fnc_qrf_areaParam)  && !( (["lastCooldownTime", _i, nil, false] call bis_fnc_qrf_areaParam) isEqualTo -1)  ) then {
				private _lastCooldownDate = ["lastCooldownTime", _i, nil, false] call bis_fnc_qrf_areaParam;
				private _elapsedTime = [(["lastCooldownTime", _i, nil, false] call bis_fnc_qrf_areaParam), date, "m"] call bis_fnc_CalculateDateTime;  // QRF_ELAPSED_TIME(_lastCooldownTime);
				private _qrfCooldown = if ( (["areaTimeout", _i, nil, false] call bis_fnc_qrf_areaParam) ) then { (GVAR(QRF, "areaCooldown")) } else { (GVAR(QRF, "qrfCooldown")) };
				private _timeoutTime = _elapsedTime >= _qrfCooldown; //QRF_TIMEOUT_TIME(_elapsedTime, _qrfCooldown);
				//[["QRF Checking cooldown[%1] %2 >= %3 ELAPSED: %4", _i, _elapsedTime, _qrfCooldown, _timeoutTime], "systemChat"] call bis_fnc_qrf_debugLogRadio;
				if (_timeoutTime) then {
					// [["QRF TIMEOUT ELAPSED"]] call bis_fnc_qrf_debugLogRadio;
					["elapsed", _i, true] call bis_fnc_qrf_areaParam;
					if ( (["areaTimeout", _i, nil, false] call bis_fnc_qrf_areaParam)) then {[_i] call bis_fnc_qrf_stopAreaMonitoring;}
				};
			};
		};
		sleep _timeoutDelay;
		/* sleep 1; */
	};
};
