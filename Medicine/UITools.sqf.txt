BIS_fnc_OM_medicine_generatePillsPool =
{
  params[ ["_pillsInPool",WARN_MILD,[99]] ];
  if(_pillsInPool <= 0) exitWith {""};
  private _pillsToShow = linearConversion [WARN_CRITICAL, WARN_MILD*2 ,_pillsInPool, 0, PILLS_POOL_MAX, true];

  private _text = "";
  for "_iLoop" from 0 to _pillsToShow-1 do
  {
    _text = _text + "<img align='right' width='16' image='a3\Missions_F_Oldman\Data\img\PhoneContacts\"+ ICON_PILLS + "'/>";
  };
  _text
};

BIS_fnc_OM_medicine_convertHealthStatus =
{
  params[ ["_villagerHealth",WARN_DYING, [999] ], [ "_age",20,[99] ] ];
  if(_villagerHealth == -1) then {_villagerHealth = 9999.0};  //if he has pills
  private _max = linearConversion [SURVIVAL_MIN_AGE, SURVIVAL_MAX_AGE, _age, SURVIVAL_MIN_AGE_TIME, SURVIVAL_MAX_AGE_TIME];
  private "_healthState";
  //not dying stage
  if(_villagerHealth > WARN_DYING ) then
  {
    _healthState = linearConversion[ WARN_DYING, _max, _villagerHealth, 0, 1, true ];
    _healthState = STATES_INFO select {_x select 0 <= _healthState && _healthState < _x select 1};
  }
  else
  {
    //dying stage
    _healthState = linearConversion[ 0, WARN_DYING, _villagerHealth, 0, 1, true ];
    _healthState = STATES_INFO_DYING select {_x select 0 <= _healthState && _healthState < _x select 1};
  };

  if(count _healthState == 0) exitWith {["n/a","n/a"]};
  //text state, icon suffix
  [_healthState select 0 select 2, _healthState select 0 select 3]
};

BIS_fnc_OM_medicine_getVillagersInfo =
{
  params[ "_villagers" ];
  private _villagersInfo = [];
  {
    private _villagerAge = [_x select 0] call BIS_fnc_OM_medicine_getPersonAge;
    private _villagerRemainingTime = [_x select 0] call BIS_fnc_OM_medicine_getRemainingTime;
    _villagersInfo = _villagersInfo + [ [_x select 0, _x select 1, _villagerRemainingTime, _villagerAge, [_villagerRemainingTime, _villagerAge] call BIS_fnc_OM_medicine_convertHealthStatus]];
  } forEach _villagers;
  _villagersInfo
};

BIS_fnc_OM_medicine_villagersStateChanged =
{
  params[ "_villagers" ];
  private _res = false;
  private _villagersPrev =  (missionNameSpace getVariable "villagersInfo");
  ["MEDICINE: _villagersPrev %1", _villagersPrev] call BIS_fnc_logFormat;
  if!(isNil "_villagersPrev") then
  {
    if(floor ([] call BIS_fnc_OM_medicine_getCurrentSupply) == floor (_villagersPrev select 0)) then
    {
      private _villagersInfo = [_villagers] call BIS_fnc_OM_medicine_getVillagersInfo;
      {
        if!(_x select 4 isEqualTo ((_villagersPrev select 1) select _forEachIndex select 4)) exitWith {["MEDICINE: Villagers %1 state changed %2", _x select 0, _x select 4] call BIS_fnc_logFormat; _res = true};
      } forEach _villagersInfo select 1;
    }
    else
    {
      /* ["MEDICINE: Medicine decreased %1", floor ([] call BIS_fnc_OM_medicine_getCurrentSupply)] call BIS_fnc_logFormat; */
      _res = true;
    };
  };
  /* ["MEDICINE: Exiting check state with %1", _res] call BIS_fnc_logFormat; */
  _res
};

BIS_fnc_OM_medicine_startShowingInfoInDiary =
{
  params
  [
      "_taskID",
      [ "_villagers",
          [
            ["BIS_Arthur",  "artur"],
            ["BIS_Lucas",   "villager_2"],
            ["BIS_Howard",  "villager_3"],
            ["BIS_John",    "villager_4"],
            ["BIS_Renly",   "villager_5"]
          ],
          [[]]
      ]
  ];

  [_villagers, _taskID] spawn
  {
    params["_villagers", "_taskID"];
    private _info =
    [
      [] call BIS_fnc_OM_medicine_getCurrentSupply,
      _villagers
    ] call BIS_fnc_OM_medicine_generatePoolInfo;
    [_taskID, _info, false] call bis_fnc_setDescription;

    while{true} do
    {
      if (BIS_OM_Outro) exitWith {};
      /* ["MEDICINE: Checking State", 0] call BIS_fnc_logFormat; */
      if( [_villagers] call BIS_fnc_OM_medicine_villagersStateChanged ) then
      {
        /* ["MEDICINE: State changed - updating description", 0] call BIS_fnc_logFormat; */
        _info =
        [
          [] call BIS_fnc_OM_medicine_getCurrentSupply,
          _villagers
        ] call BIS_fnc_OM_medicine_generatePoolInfo;
        /* ["MEDICINE: _text: %1", _info] call BIS_fnc_logFormat; */
        [_taskID, [_info], false] call bis_fnc_setDescription;
        Sleep 2;
      };
      Sleep 10;
    };
  };
};
