BIS_fnc_OM_openTheStrategicMap =
{
    params ["_logic"];

    private _eventHandlers = [];
    SWITCH_TO_TARGESEL_MODE();
    private _activeSectors = (nearestObjects [player, ["ModuleOMSector_F"], 250]);
		_logic setVariable ["activeSector", if(count _activeSectors > 0) then {_activeSectors select 0}];  //TODO: once we have the onSpawn Script fired when the sector is spawned, use a flag rather than nearestobject

		["SYNDIKATTEAM: active sectors: %1", _activeSectors] call BIS_fnc_logFormat;

    if (!(isNil {(missionNameSpace getVariable "BIS_OM_customEventHandlers" select 0)})) then { [ EVENT_DESTROY, uiNamespace getVariable "BIS_OM_SyndikatTeam_ID" ] call BIS_fnc_OM_handleMapControls; };	//TODO: replace 0 with parameter

    private _sectors = [] call BIS_fnc_OM_BuildSectorsData;
    SVAR( _logic,  "sectors", _sectors );

		["SYNDIKATTEAM: sectors: %1", _sectors] call BIS_fnc_logFormat;

    uiNamespace setVariable
		[
			"BIS_OM_SyndikatTeam_ID",
			[ EVENT_INIT, [],
        {
          [ MODE_CREATE_ASSAULT, [_this select 2, _this] ] call BIS_fnc_OM_ModuleSyndikatTeam;
        }, EVENT_MOUSE_BUTT_DOWN
      ] call BIS_fnc_OM_handleMapControls
		];

    if(!visibleMap) then { openMap true };
		["SYNDIKATTEAM: opening map: %1", 1] call BIS_fnc_logFormat;
    /* private _eventInfo = "<t align='left'><t size='1.25' color='#ff0000'>Welcome to the Syndikat Camp</t><br/>";
    _eventInfo = _eventInfo + "<t size='1.00' color='#71a0a4'>Open the map</t><br/>";
    _eventInfo = _eventInfo + "<t size='1.00' color='#71a0a4'>and select the base</t><br/>";
    _eventInfo = _eventInfo + "<t size='1.00' color='#89bd35'>you want to attack.</t><br/></t>";
    //hint _eventInfo;
    hintSilent parseText format
    [
      _eventInfo, 0, 0, 0, 0
    ];
		*/
		[] spawn
		{
			waitUntil{ Sleep 1; !visibleMap };
			["SYNDIKATTEAM: deactivating camp: %1", 1] call BIS_fnc_logFormat;
			[ MODE_DEACTIVATE_CAMP ] call BIS_fnc_OM_ModuleSyndikatTeam;
		};
};
