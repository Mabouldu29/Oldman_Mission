	private _prevModuleName = _module getVariable "_moduleNameCopy";

		private _building = _buildings#_i;
		private _bName = (_building get3DENAttribute "name")#0;

		["quest DESTROY SET_NAMES TEST %1 %2 %3 %4 %5 %6",_bName select [0, count _moduleName], _bName select [0, count _prevModuleName], _moduleName, _prevModuleName ,(((_bName select [0, count _moduleName]) isEqualTo _moduleName) ||   ((_bName select [0, count _prevModuleName]) isEqualTo _prevModuleName
		
		)) ] call bis_fnc_logFormat;

		 ((_bName select [0, count _moduleName]) isEqualTo _moduleName) ||

		) then {
			["quest DESTROY SET_NAMES RESET %1 %2",_building, _building set3DENAttribute ["name", ""] ] call bis_fnc_logFormat;
		} else {

			};
			_buildingNames pushBack _bName;

			// ["quest DESTROY NNED DEL %1", [_bName isEqualTo _buildingTargetName, _bName, _buildingTargetName] ] call bis_fnc_logFormat;
			//["quest DESTROY NNED DELETE NAME %1",] call bis_fnc_logFormat;

		// ["quest DESTROY NNED %1 - %2", _bName, _building] call bis_fnc_logFormat;

private _drawEhRemove = {

		{
			removeMissionEventHandler ["Draw3D",_ehDraw3D];
			missionNamespace setVariable ["resourceModules_ehDraw3D",-1];
		};
	};
	
	private _buildings = _module getVariable ["#buildings", []];

				_building setVariable["#marriedTo", objNull];
			} forEach _buildings;
		} forEach _buildings;

private _useTerrainObject = {
	params["_module", ["_displayErrors", false], ["_buildings", []], ["_terrainBuildingRadius", 10], ["_appendObjects", false] ];


	private _appended = [];
	if (_appendObjects) then {
		_appended = _buildings;
	};	

	if (_buildings isEqualTo [] || _appendObjects) then { //!(_terrainBuildingRadius isEqualTo -1) && 
		
		private _types = (_module getVariable ["AllowedObjectClassName", ""]) splitString ",";
		["quest DESTROY APPEND %1", _types] call bis_fnc_logFormat;
		
		if (count _types > 0) then {
			_buildings = [_module, _terrainBuildingRadius, _types] call _getTerrainBuildingsEden;
			_buildings append _appended;
		};
	};

	//TODO POTENTIALY BREAKING FOR THE OTHER USAGE
	if (!_appendObjects && !(_terrainBuildingRadius isEqualTo -1)) then {
		 _buildings = [_module, _buildings, _displayErrors] call _getTerrainBuildingEden;	
		_buildings append _appended;
	};
	
	private _buildingsRes = _buildings;
	if (count _buildings > 0) then {  
		if ([_module] call _isSingle) then {
			_buildingsRes = _buildings#0;
			// ["quest DESTROY USE single %1", _buildingsRes] call bis_fnc_logFormat;	
			_module setVariable ["#buildings", [_buildingsRes]];
		} else {
			// ["quest DESTROY USE multi %1", _buildingsRes] call bis_fnc_logFormat;	
			_module setVariable ["#buildings",_buildingsRes];
		};
		_module call _setTerrainBuildingEden;
	} else {
		// ["quest DESTROY USE NO %1", ""] call bis_fnc_logFormat;	
		_module setVariable ["#buildings", [] ];
	}; 

	// if (!((_editorBuildingName#0) isEqualType []) then {

		
	// };
	_buildingsRes
};

		{
			private _buildingCandidate = _x;
			private _buildingType = typeOf _buildingCandidate;
			
			["quest DESTROY ONE_OBJECT"] call bis_fnc_logFormat;

			if (_buildingType != "") then {
				private _b = configFile >> "cfgVehicles" >> _buildingType;
				if (getText(_b >> "destrType") == "DestructNo" || getText(_b >> "destrType") == "") then { //TODO some buildings could have DestructionEffects>>Ruin1 set to same model
					private _messageIndestructable = format ["This object is indestructible (%1'%2') !", _buildingType, getText(_b >> "displayName")];
					["quest DESTROY ONE_OBJECT ERR %1", _messageIndestructable] call bis_fnc_logFormat;
					if (_displayErrors) then {
						["Error: %1", _messageIndestructable] call BIS_fnc_error;
					};
				} else {
					_buildingsResult pushBack _buildingCandidate;
				};	  
			} else {
				_buildingsResult pushBack _buildingCandidate;
			};
		} forEach _buildings;
	};	
	// ["quest DESTROY GET_TB RESULT %1", _buildingsResult] call bis_fnc_logFormat;
	_buildingsResult
};

	private _buildings =  [_buildings, _displayErrors] call _getTerrainBuilding;
