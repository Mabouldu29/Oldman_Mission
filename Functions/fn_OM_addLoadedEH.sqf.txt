	}

addMissionEventHandler 
[
	"Loaded",
	{
		scriptName "addMissionEventHandler/Loaded(loadingScreen)";
		uiNamespace setVariable ["BIS_RscDisplayLoadingScreen_fromLoad", true];			//tell the loadingscreen the LOAD event has happened, therefore we need to use the another method for animations;
		endLoadingScreen;
		startLoadingScreen ["","RscDisplayLoadingOM"];
		[] spawn
		{
			BIS_OM_preloadHandler = addMissionEventHandler
			["PreloadFinished",
				{
					scriptName "addMissionEventHandler/PreloadFinished(loadingScreen)";
					["LOADINGSCREEN:  setting to false %1", 1] call BIS_fnc_logFormat;
					uiNamespace setVariable ["BIS_RscDisplayLoadingScreen_fromLoad", false];
					uiNamespace setVariable ["BIS_RscDisplayLoadingScreen_fromLoaded", false];
					removeMissionEventHandler ["PreloadFinished", BIS_OM_preloadHandler];
					endLoadingScreen;
				}
			];
		};
	}
];

addMissionEventHandler
[	
	"Loaded",
	{
		params ["_saveType"];
		[] spawn
		{
			scriptname "fn_OM_HandleMapControls.sqf: Loaded";
			waituntil {!isnull finddisplay 12};
			if (!(isNil {uiNamespace getVariable "BIS_OM_HandleMapControls_customEventHandlers"})) then
			{
				{
					private _eventHandlers = _x;
					if( count _eventHandlers == 0 ) exitWith {};
					{
							disableSerialization;
							// ["MAPHANDLER: Removing eventhandler: %1 | %2", _eventHandlers select 1 select 0 select 0, _eventHandlers select 1 select 0 select 1] call BIS_fnc_logFormat;
							(UINamespace getvariable "BIS_OM_HandleMapControls_mapHandler") ctrlRemoveEventHandler [_eventHandlers select 1 select 0 select 0, _eventHandlers select 1 select 0 select 1];
					} forEach _eventHandlers;
				} forEach (uiNamespace getVariable "BIS_OM_HandleMapControls_customEventHandlers");
				UINamespace setvariable [ "BIS_OM_HandleMapControls_customEventHandlers", [] ];
				missionNameSpace setVariable ["BIS_OM_HandleMapControls_customCallbacks", [] ];
			};
		}
	}
];
