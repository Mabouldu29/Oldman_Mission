// EH to remove picked items from composition.
[] spawn 
{
	waitUntil {!isNull player};
	player addEventHandler ["Take",
	{
		params ["_unit", "_container"];
		private _sector = _container getVariable ["#sector", objNull];
		if (!isNull _sector) then
		{
			private _objectsRef = _sector getVariable ["#objectsRef", []];
			private _index = _objectsRef find _container;
			if (_index >= 0) then
			{
				// remove picked item from composition
				private _objectData = _sector getVariable ["#objectData", []];
				private _item = _objectData select _index;
				_item set [0, ""];
				assert (count _objectsRef == count _objectData);
			};
		};
	}];
};
