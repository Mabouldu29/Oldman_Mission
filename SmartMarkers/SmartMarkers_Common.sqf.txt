BIS_fnc_OM_moduleSmartMarkers_SetMarkerVisibility =
{
	/* private _scr = format ["scr_%1", time*1000]; */
  params ["_marker", "_finalAlpha", "_duration", "_actualAlpha"];

	_finalAlpha = ROUND(_finalAlpha);
	_duration = ROUND(_duration);

  private _effectsInProgress = (missionNameSpace getVariable ["BIS_OM_SmartMarkers_mrkHndl", []]);

  if({_x isEqualTo _marker} count _effectsInProgress >= 2) exitWith {true};
  _effectsInProgress pushBack _marker;
	private _t = time;
  WaitUntil { {_x isEqualTo _marker} count _effectsInProgress == 1 || time > _t + MARKER_INTERRUPT_DELAY};

  private _startAlpha = ROUND(markerAlpha _marker);
  private _alpha = _startAlpha;
	private _difference = _startAlpha - _finalAlpha;
	if(_difference != 0) then
	{
		/* ["difference: %1", _difference] call BIS_fnc_logFormat; */
		private _nrOfSteps = abs (round linearConversion[0,1,_difference,1,MARKER_ANIMATION_FRAMES]);
		/* ["nrOfSteps: %1", _nrOfSteps] call BIS_fnc_logFormat; */
		if(_nrOfSteps == 0) then {_nrOfSteps = 1};
	  private _step = _difference/_nrOfSteps;
		/* ["_step: %1", _step] call BIS_fnc_logFormat; */
	  private _dt = abs( _duration / _nrOfSteps );
		/* ["_dt: %1", _dt] call BIS_fnc_logFormat; */

	  while { OP(_alpha,_finalAlpha) && ({_x isEqualTo _marker} count _effectsInProgress == 1)} do
	  {
			_alpha = ROUND(_alpha - (_step));
	    _marker setMarkerAlpha _alpha;
	    /* ["%2 _alpha = %1", _alpha, _scr] call BIS_fnc_LogFormat; */
	    if({_x isEqualTo _marker} count _effectsInProgress > 1) exitWith {false};
	    sleep _dt;
	  };
	};
  _effectsInProgress deleteAt ( _effectsInProgress findIf {_x isEqualTo _marker} );
};

BIS_fnc_OM_moduleSmartMarkers_SetLocatorPosition =
{
  params["_map", "_smartMarkerBase", "_color"];
  private _marker = createMarker [MARKER_NAME_UNIQUE("LOCATOR"), [0,0,0]];
	_marker setMarkerShape "ICON";
	_marker setMarkerType LOCATOR_MARKER;
	_marker setMarkerSize LOCATOR_MARKER_SIZE;
	_marker setMarkerColor _color;
	_smartMarkerBase setVariable ["markerLocator", _marker];
  private _alpha = 0.0;
  while{_smartMarkerBase getVariable SMARTMARKER_HIGHLIGHTED} do
  {
    if(visibleMap) then   //&& in the map range
    {
      private _smartMarkerBaseScreenPos = _map posWorldToScreen position _smartMarkerBase;
      private _angle = 90;
      private _x = _smartMarkerBaseScreenPos select 0;
      private _y = _smartMarkerBaseScreenPos select 1;

      if!(IS_IN_SAFEZONE_X( _smartMarkerBaseScreenPos select 0)) then
      {
        _smartMarkerBase setVariable ["outside", true];
        if(_smartMarkerBaseScreenPos select 0 > (safezoneX + safezoneW)) then
        {
          _x = (safezoneX + safezoneW)-0.027;
        }
        else
        {
          _x = safezoneX+0.027;
          _angle = 270;
        };
        _marker setMarkerPos (_map posScreenToWorld [_x, _y]);
        _marker setMarkerDir _angle;
      };
      if!(IS_IN_SAFEZONE_Y( _smartMarkerBaseScreenPos select 1)) then
      {
        _smartMarkerBase setVariable ["outside", true];
        if(_smartMarkerBaseScreenPos select 1 > (safezoneY + safezoneH)) then
        {
          _y = (safezoneY + safezoneH)-0.027;
          _angle = 180;
        }
        else
        {
          _y = safezoneY+0.10;
          _angle = 0;
        };
        _marker setMarkerPos (_map posScreenToWorld [_x, _y]);
        _marker setMarkerDir _angle;
      };
      if( (IS_IN_SAFEZONE_X( _smartMarkerBaseScreenPos select 0)) && (IS_IN_SAFEZONE_Y( _smartMarkerBaseScreenPos select 1)) ) then
      { _alpha = 0.0; _smartMarkerBase setVariable ["outside", false]; _marker setMarkerAlpha _alpha;};

    };
    Sleep 0.02;
  };
  deleteMarker _marker;
};

	params ["_smartMarker", "_system", "_descr"];
	private _descriptions = _smartMarker getVariable [ SMARTMARKER_EXTRADATA, [] ];
	_descriptions pushBack [count _descriptions, _system, _descr];
};

BIS_fnc_OM_moduleSmartMarkers_RemoveExtraDescription =
{
	params ["_smartMarker", "_descrID"];
	private _descriptions = _smartMarker getVariable [ SMARTMARKER_EXTRADATA, [] ];
	private _index = _descriptions findIf {_x select 0 == _descrID};
	if( _index != -1 ) then
	{
		_descriptions deleteAt _index;
	};
};

BIS_fnc_OM_moduleSmartMarkers_RemoveExtraDescriptionBySystemID =
{
	params ["_smartMarker", "_systemID"];
	private _descriptions = _smartMarker getVariable [ SMARTMARKER_EXTRADATA, [] ];
	_smartMarker setVariable [ SMARTMARKER_EXTRADATA, _descriptions select { (_x select 1) != _systemID } ];
};
