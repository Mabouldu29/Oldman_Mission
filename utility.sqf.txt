BIS_fnc_numberToSide = {
	scriptName "BIS_fnc_numberToSide";
	params["_number"];
	private _side = sideUnknown;
	if (_number isEqualType sideUnknown) exitWith {_number};
	private _sides = [EAST,WEST,RESISTANCE,CIVILIAN,sideUnknown,sideEnemy,sideFriendly,sideLogic,sideEmpty,sideAmbientLife];

BIS_fnc_om_randomPosCode = {
	scriptName "BIS_fnc_om_randomPosCode";
	params ["_area",["_code",{true}],["_maxRepeats",40]];
	_area params ["_pos","_px","_py"];
	_pos params ["_posX","_posY"];
	private _range = _px max _py;
	private _resultPos = [];
	private _positions = [];

	// Prepare random locations
	for "_i" from 0 to 19 do {
		_positions pushBack [(_range * random 2) - _range + _posX,(_range * random 2) - _range + _posY];
	};
	// Filter out only positions in area
	_positions = _positions inAreaArray _area;

	// Check code
	private _i = _positions findIf _code;
	if (_i > -1) exitWith {
		(_positions#_i) pushBack 0;
		(_positions#_i)
	};

	// Repeat version
	for "_j" from 0 to ceil (_maxRepeats / 20) do {
		private _positions = [];
		for "_i" from 0 to 19 do {
			_positions pushBack [(_range * random 2) - _range + _posX,(_range * random 2) - _range + _posY];
		};
		// Filter out only positions in area
		_positions = _positions inAreaArray _area;

		// Check code
		private _i = _positions findIf _code;
		if (_i > -1) exitWith {_resultPos = _positions#_i};
	};
	_resultPos pushBack 0;
	if (count _resultPos != 3) then {_resultPos = [_pos # 0, _pos # 1, 0]};
	_resultPos
};

// Handle all cases of skipped time
BIS_fnc_om_handleTimeSkip = {
	scriptName "BIS_fnc_om_handleTimeSkip";
	params ["_time"];
	private _threatToDelete = _time * 25;

	// Handle relationship decrease
	private _threat = [east] call BIS_fnc_om_threat;
	[east, _threatToDelete min _threat] call BIS_fnc_om_lowerThreat; // Lower threat by seconds resting, or to 0

	// Day/night change
	private _changed = false;
	private _wakeUpTime = daytime + _time;

	if (_wakeUpTime >= 24) then {_wakeUpTime = _wakeUpTime - 24};
	(date call BIS_fnc_sunriseSunsetTime) params ["_dawn","_dusk"];
	if (_wakeUpTime > _dawn && _wakeUpTime < _dusk) then {
		if (BIS_night) then {
			BIS_night = false;
			_changed = true;
		};
	} else {
		if (!BIS_night) then {
			BIS_night = true;
			_changed = true;
		};
	};

	if (_changed) then {
		// Despawn all automatically handled sectors, they will respawn on their own
		[] spawn {
			scriptName "utility/despawnSector";
			sleep 0.1;
			private _playerSector = [getPos player] call BIS_fnc_om_inSector;
			if (!isNull _playerSector) then {
				[_playerSector,true] spawn BIS_fnc_om_sectorDeactivate;
			};
			{
				if (_x != _playerSector) then {
					[_x,true] spawn BIS_fnc_om_sectorDeactivate;
				};
			} forEach (BIS_allSectors select {GVAR(_x,"#state" >= 2) && !GVARS(_x,"#manual",false)});
		};
	};
};
