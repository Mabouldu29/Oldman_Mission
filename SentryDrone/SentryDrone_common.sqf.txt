BIS_fnc_OM_ModuleSentryDrone_ScanArea =
{
  params ["_object", "_from", "_to", "_step", ["_distance", 15, [999]], ["_axis", 0, [9] ] ];
  private _nearestObject = objNull;
  private _realNearestObject = [0,0,0];
  for "_iLoop" from _from to _to step _step do
  {
    _nearestObject = [_object, [_iLoop,0,0], _distance, _axis] call BIS_fnc_OM_ModuleSentryDrone_IntersectInDirection;
    _nrOfObjects = count _nearestObject;
    if( _nrOfObjects  > 0 ) then
    {
      _nearestObject = _nearestObject select (_nrOfObjects - 1);
      if(_nearestObject distance2D _object < _realNearestObject distance2D _object) then
      {
        _realNearestObject = _nearestObject;
      };
    };
    Sleep 0.1;
  };
  if(typeName _realNearestObject == "ARRAY") then
  {
    [_realNearestObject, -1]
  }
  else
  {
    [_realNearestObject, [_realNearestObject] call BIS_fnc_OM_ModuleSentryDrone_GetSafeHeightATL]
  };
};

BIS_fnc_OM_ModuleSentryDrone_ScanAreaCone =
{
  params["_object", ["_distance", 15, [999]], ["_direction", 1, [9]] ];
  private _scannedObjects = [];
  private _nearestObject = objNull;
  private _realNearestObject = [0,0,0];
  private _startZ = 45; private _endZ =  -45; private _stepZ = -5; private _iZ = _startZ;
  private _startX = 90; private _endX =  0; private _stepX = 1*_stepZ; private _iX = _startX;
	private _iX2 = 9;
  while {_iZ > 0} do
  {
		_iX = 40.961*ln _iX2;
		_nearestObject = [_object, [_iX,0,_iZ], _distance] call BIS_fnc_OM_ModuleSentryDrone_IntersectInDirection;
    _iZ = _iZ + _stepZ;
		_iX2 = _iX2 - 1;
	   _scannedObjects pushBack _nearestObject;
    Sleep 0.2;
  };
  _iZ = 0; _iX = 0;_iX2 = 1;

  while {_iZ >= _endZ} do
  {
		_iX = 40.961*ln _iX2;
		_nearestObject = [_object, [_iX,0,_iZ], _distance] call BIS_fnc_OM_ModuleSentryDrone_IntersectInDirection;
    _iZ = _iZ + _stepZ;
		_iX2 = _iX2 + 1;
    _scannedObjects pushBack _nearestObject;
    Sleep 0.2;
  };
  private _lastDistance = 99999;
  {
    private _arr = _x;
    private _sizeOfArray = count _arr;
    if(_sizeOfArray > 0) then
    {
      private _nearestObjectActual = _arr select (_sizeOfArray - 1);
      //["_nearestObjectActual: %1"] call BIS_fnc_logFormat;
      private _actualDistance = _nearestObjectActual distance2D _object;
      if(_actualDistance < _lastDistance) then
      {

        _realNearestObject = _nearestObjectActual;
        //["_realNearestObject: %1"] call BIS_fnc_logFormat;
        _lastDistance = _actualDistance;
      };
    };
  } forEach _scannedObjects;
  if(typeName _realNearestObject == "ARRAY") then
  { [_realNearestObject, -1] } else { [_realNearestObject, [_realNearestObject] call BIS_fnc_OM_ModuleSentryDrone_GetSafeHeightATL] }
};

BIS_fnc_OM_ModuleSentryDrone_MoveInSafeHeight2 =
{
  params [ ["_sourceObject", objNull, [objNull]] ];
  //missionNameSpace setVariable ["sourceObject", _sourceObject];
  _sourceObject setVariable ["BIS_flyInSafeHeight", true];

  [_sourceObject] spawn
  {
    params ["_sourceObject"];
    private _obstacle = [_sourceObject, 8] call BIS_fnc_OM_ModuleSentryDrone_ScanAreaCone;
    /* str _obstacle call BIS_fnc_Log; */
    private _safeAltitude = _obstacle select 1;

    if(_safeAltitude == -1) then {_safeAltitude = 5};

    _sourceObject setVariable ["nearestObstacle", _obstacle select 0];
    _sourceObject setVariable ["nearestObstacleDistance", (_obstacle select 0) distance2D _sourceObject];
    _sourceObject setVariable ["safeHeight", _safeAltitude];
    _sourceObject flyInHeight _safeAltitude;
  };
};

BIS_fnc_OM_ModuleSentryDrone_MoveInSafeHeight =
{
  params [ ["_sourceObject", objNull, [objNull]] ];
  missionNameSpace setVariable ["sourceObject", _sourceObject];
  _sourceObject setVariable ["BIS_flyInSafeHeight", true];

  while {alive _sourceObject && _sourceObject getVariable "BIS_flyInSafeHeight"} do
  {
    if((_sourceObject distance2D (waypointPosition (waypoints _sourceObject select (currentWaypoint group _sourceObject)))) < 20) then
    {
      [_sourceObject] spawn
      {
        params ["_sourceObject"];
        /* private _obstacle = [_sourceObject, 0,-90,-10,50] call BIS_scanArea; */
        private _obstacle = [_sourceObject, 8] call BIS_fnc_OM_ModuleSentryDrone_ScanAreaCone;
        /* private _obstacle = [_sourceObject, 80, 90, 10, 15] call BIS_fnc_OM_ModuleSentryDrone_ScanArea; */
        str _obstacle call BIS_fnc_Log;
        private _safeAltitude = _obstacle select 1;

        if(_safeAltitude == -1) then {_safeAltitude = 5};

        _sourceObject setVariable ["nearestObstacle", _obstacle select 0];
        _sourceObject setVariable ["nearestObstacleDistance", (_obstacle select 0) distance2D _sourceObject];
        _sourceObject setVariable ["safeHeight" , _safeAltitude];
        _sourceObject limitSpeed 12;
        _sourceObject flyInHeight _safeAltitude;
      };
    }
    else
    {
      _sourceObject flyInHeight 16;
      _sourceObject limitSpeed 350;
			[_sourceObject] call BIS_fnc_OM_ModuleSentryDrone_HideVector;
    };
  Sleep 1.5;
  };
};

BIS_fnc_OM_ModuleSentryDrone_MoveActualWaypointSet =
{
  params ["_groupFrom", "_groupTo", ["_restore", false, [false]] ];

  if(isNil "_groupTo") then
  {
    private _group = createGroup [CIVILIAN, true];
    _groupTo = _group createUnit ["Logic", [0,0,100], [], 0, "FORM"]
  };
  private _groupWPs = waypoints group _groupFrom;
  group _groupTo copyWaypoints group _groupFrom;

  if(_restore) then
  {
    _groupFrom = _groupTo getVariable "BIS_WPStorage";
    group _groupTo setCurrentWaypoint (_groupFrom getVariable "BIS_lastCurrentWP");
    deleteVehicle _groupFrom;
  }
  else
  {
    _groupFrom setVariable ["BIS_WPStorage", _groupTo];
    if(group _groupFrom != grpNull && {count _groupWPs != 0}) then
    { _groupTo setVariable ["BIS_lastCurrentWP", _groupWPs select (currentWaypoint group _groupFrom)] };
  };

  deleteWaypoint [group _groupFrom, (waypoints _groupFrom findif {waypointType _x == "CYCLE"})];
  private _nrOfWPs = count _groupWPs;

  for "_iLoop" from 0 to _nrOfWPs do
  {
    [group _groupFrom,0] setWaypointPosition [position _groupFrom, 0];
    deleteWaypoint [group _groupFrom,0]
  };

};

BIS_fnc_OM_ModuleSentryDrone_ReturnDrone =
{
  params ["_wp", "_originalWp", "_drone", "_target", "_originalDronePosition", "_originalAltitude", "_message"];
  if(GVAR(_drone, "droneState") == DRONE_STATE_HUNTING) then
  {
    [_drone getVariable "BIS_WPStorage", _drone, true] call BIS_fnc_OM_ModuleSentryDrone_MoveActualWaypointSet;
    _drone forgetTarget _target;
    _drone flyInHeight _originalAltitude;
    _message call BIS_fnc_log;
    SVAR(_drone "droneState", DRONE_STATE_IDLE);
  };
};

BIS_fnc_OM_ModuleSentryDrone_PopAndPushWP =
{
  /* PRIORITY_QUEUE_DEBUG = true; */
  params ["_drone", "_hunted"];
  private _wpQueue = _drone getVariable "BIS_wpQueue";
  private _markersArray = _drone  getVariable ["BIS_WPmarkers", []];
  if([_wpQueue] call BIS_fnc_PriorityQueue_IsFull) then
  {
    [_wpQueue, _drone, _markersArray] call BIS_fnc_OM_ModuleSentryDrone_DeleteWP;
  };
  [_wpQueue, _drone, _hunted, _markersArray] call BIS_fnc_OM_ModuleSentryDrone_AddWP;
};

BIS_fnc_OM_ModuleSentryDrone_AddWP =
{
  params ["_wpQueue","_drone", "_target", "_markersArray"];
  private "_center";
  if(typeName _target == "OBJECT") then
  { _center = getpos _target };
  private _newWP = createTrigger ["EmptyDetector", _center];
  _newWP setTriggerArea [4, 4, 0, false];
  _newWP setTriggerActivation ["EAST", "PRESENT", true];
  _newWP setTriggerStatements ["this", "(thislist select 0) setVariable ['BIS_activatedTrigger', thisTrigger]; (thislist select 0) setVariable ['BIS_droneState_fsm1',4]", ""];

  private _prio = 0;
  if!([_wpQueue] call BIS_fnc_PriorityQueue_IsEmpty) then
  {
    _prio = [_wpQueue] call BIS_fnc_PriorityQueue_GetLowestPriority;
  };
  [_wpQueue, _newWP, _prio-1] call BIS_fnc_PriorityQueue_PushItem;
  //_markersArray pushBack ( [ "hd_dot", [], _center, false ] call BIS_fnc_markerCreate );
  //_drone setVariable ["BIS_WPmarkers", _markersArray];
  _newWP
};
