BIS_fnc_OM_ModuleEconomy_getEquipmentType =

	private _found = false;
	{
		if( toLower configName (configFile >> _x >> _equipmentClass) isEqualTo toLower _equipmentClass ) then
		{
			_equipType = [(configFile >> _x >> _equipmentClass),"type"] call BIS_fnc_returnConfigEntry;
			if(isNil "_equipType") then {_equipType = WeaponSlotUnknown};
		};
		if(!isNil "_equipType") exitWith {_equipType};
	} forEach CFGS;

	private _ret = [];
	/* if(typeName _equipType == "STRING") then {call compile format ["_equipType = %1", _equipType]}; */
	if(!isNil "_equipType") then
	{
		if(_equipType isEqualType "") then {_equipType = WeaponSlotItem};

		if(_equipType == 0) then
		{
			_ret pushBack 0;
		}
		else
		{
			{
				if(LOG_AND_DECIMAL(_equipType, _x) == _x) then {_ret pushBack _x};
			} forEach SLOT_TYPES;
		};
	};
	_ret
};

BIS_fnc_OM_ModuleEconomy_getEquipmentPrice =
{
	/*
		The final price depends on the following factors:
	 		• cost / purchase - if player buys or sells the item
	 		• reputation - better == lower cost price and higher selling price
	 		• purchase factor - how much the shopkeeper is interested in the item player will try to sell him
	*/
	PROFILE_INFO("getEquipmentPrice","getEquipmentPrice",1);
	params[ ["_equipmentClass","",[""] ], [ "_purchasePrice", false, [false] ] ];
	/* if(typeName _equipmentClass == "ARRAY") then {_equipmentClass == _equipmentClass select  */
	private _price = [_equipmentClass] call BIS_fnc_OM_ModuleEconomy_getEquipmentPriceFromList;
	if( !isNil "_price" ) exitWith
	{
		_price = (CALCULATE_DISCOUNT(_price, _purchasePrice));
		if(_purchasePrice) then
		{
			_price = _price * ([_equipmentClass] call BIS_fnc_OM_ModuleEconomy_getEquipmentPurchaseFactor);
		};
		_price
	};
	private _equipmentType = [_equipmentClass] call BIS_fnc_OM_ModuleEconomy_getEquipmentType;
	if(!isNil "_equipmentType" && {count _equipmentType > 0}) then
	{
		{ if(_x#0 == (_equipmentType#0)) exitWith { CALCULATE_DISCOUNT(_x#1, _purchasePrice) } } forEach EQUIP_PRICES;
	}
	else
	{
		""
	};
};

{
	params["_number", ["_size",8,[99] ] ];

	private _remainder = 0;
	private _result = [];
	for "_iLoop" from 1 to _size do
	{ _result = _result + [0]	};

	{
		_remainder = _number % 2;
		_number = floor (_number / 2);
		if(_remainder != 0) then {_result set [_size - _forEachIndex - 1, 1]};
	} forEach _result;
	_result
};
