#include "\a3\Missions_F_Oldman\Systems\defines\commonDefines.inc"
#include "definitions.inc"


//definitions for the sectors array structure

params [ ["_mode", "init", ["", 999]], ["_input", [], [[]] ] ];
_input params [ "_param1", ["_param2", 0, [999, objNull, [], "", true]]  ];
private "_logic";


if(typename _mode == "STRING" && {_mode == MODE_INIT}) then
{
	_logic = _param1;
	missionNamespace setVariable ["BIS_OM_Economy_logic", _logic];
}
else
{
	_logic = missionNamespace getVariable  "BIS_OM_Economy_logic";
};

switch _mode do
{
	case MODE_INIT:
	{
		WaitUntil{!isNil "BIS_missionInitDone"};
		[true, ""] call BIS_fnc_OM_loadingScreenWrapper;
		/* ["OM_Economy","RscDisplayLoadingOM"] call bis_fnc_startloadingscreen; */
		"BOOT: Economy init" call BIS_fnc_Log;
		#include "Economy_Init.sqf"
		[false, ""] call BIS_fnc_OM_loadingScreenWrapper;
		/* "OM_Economy" call bis_fnc_endloadingscreen; */
		"BOOT: Economy init end" call BIS_fnc_Log;
	};
	case MODE_INCREASE_MONEY:
	{
		[_logic, _param1, _input#1] call BIS_fnc_OM_moduleEconomy_IncreaseMoney;
	};
	case MODE_REDUCE_MONEY:
	{
		[_logic, _param1, _input#1] call BIS_fnc_OM_moduleEconomy_DecreaseMoney;
	};
  case MODE_GET_STATUS:
  {
		[_logic] call BIS_fnc_OM_moduleEconomy_GetStatus;
  };
  case MODE_CHANGE_INCOME:
  {
		[_logic, _param1, _input#1] call BIS_fnc_OM_moduleEconomy_ChangeIncome;
  };
	case MODE_BUY_EQUIP:
  {
		[_logic, _param1, _input#1] call BIS_fnc_OM_moduleEconomy_BuyEquipment;
  };
	case MODE_HAS_ENOUGH_MONEY:
	{
		[_logic, _param1] call BIS_fnc_OM_moduleEconomy_HasEnoughMoney;
	};
	case MODE_SELL_EQUIP:
	{
		[_logic, _param1, _input#1] call BIS_fnc_OM_moduleEconomy_SellEquipment;
	};
	case MODE_GET_EQUIP_PRICE:
	{
		[_param1, _input#1] call BIS_fnc_OM_ModuleEconomy_GetEquipmentPrice;
	};
	case MODE_PAY:
	{
		[_logic, _param1, _input#1] call BIS_fnc_OM_ModuleEconomy_Pay;
	};
};
