#include "\a3\Missions_F_Oldman\Systems\defines\questDefines.inc"
#include "\A3\Missions_F_Oldman\Systems\defines\commonDefines.inc"

if (missionNamespace getVariable ["BIS_OM_waitInProgress", FALSE]) exitWith {};

params[ ["_forcedSleepTime", -1, [9999]] ];	//if positive number is passed, then skip the dialog and put player into the sleep mode;

if( _forcedSleepTime >= 0 ) exitWith
{
	BIS_TimeToSkipWhileSleeping = _forcedSleepTime;
	BIS_SleepTimeSelected = true;
	[] spawn BIS_fnc_om_GoSleep;
};

//--- cancel function if player is in vehicle
if (vehicle player != player && (abs (speed vehicle player)) > 0) exitwith
{
	//--- show notification
	SHOW_SLEEP_TRAVEL_NOTIFICATION( Localize "STR_A3_OM_System_Rest_WarningMoving");
	_performAction = false;
};

private _performAction = true;
//--- if enemies nearby
if( count ((player nearEntities ["Man", 150]) select {side _x == east && alive _x}) > 0 ) then	//TODO: do it better, player can be threat to Syndikats as well
{
	//--- and it is not the travel dialog
	if( missionNamespace getVariable ["BIS_OM_waitDialogMode", "wait"] != "travel" ) then
	{
		//--- and player is not in the Safehouse
		if!( player getvariable ["AtGasStation",false] ) exitWith
		{
			//--- show notification and do not allow to rest/sleep
			SHOW_SLEEP_TRAVEL_NOTIFICATION( Localize "STR_A3_OM_System_Rest_WarningEnemy");
			_performAction = false;
		};
	}
	else
	{
		//--- if it is the travel dialog, allow travelling only in case player is not hunted
		if(  { [_x] call BIS_fnc_om_aggroMode == M_AGRESSIVE } count [ EAST, RESISTANCE ] > 0 ) exitWith
		{
			//--- show notification and do not allow to travel
			SHOW_SLEEP_TRAVEL_NOTIFICATION( Localize "STR_A3_OM_System_Rest_WarningTravel");
			_performAction = false;
		};
	};
};

if!(_performAction) exitWith{};

//--- wait till player release space bar!
waitUntil {inputAction "action" == 0};

player setVariable ["BIS_playerSavingUsingBag", false];
if(	((magazines player) findif {_x in SLEEPING_BAGS}) != -1 && !(player getvariable ["AtGasStation",false])	) then
{
	player setVariable ["BIS_playerSavingUsingBag", true];
};

//--- create dialog
BIS_OM_waitDialogMode = if( missionNamespace getVariable ["BIS_OM_waitDialogMode", "wait"] == "travel" ) then
{
	private _prevMode = missionNamespace getVariable ["BIS_OM_waitDialogMode", "wait"];
	missionNamespace setVariable ["BIS_OM_waitDialogMode", _prevMode];
	"travel"
}
else
{
	"wait"
};

_null = (findDisplay 46) createDisplay "RscDisplayOMRest";
//--- reset time value in counter to default
//BIS_TimeToSkipWhileSleeping = 1/3;
BIS_SleepingTime = [Nil,1/3];
